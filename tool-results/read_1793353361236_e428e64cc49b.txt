    76→    76→const DDG_UA =

    83→    83→export async function searchWebDuckDuckGo(query: string, num = 10): Promise<RawSearchItem[]> {
    84→    84→  try {
    85→    85→    const res = await fetch(
    86→    86→      `https://html.duckduckgo.com/html/?q=${encodeURIComponent(query)}`,
    87→    87→      {
    88→    88→        headers: {
    89→    89→          "User-Agent": DDG_UA,
    90→    90→          Accept: "text/html,application/xhtml+xml",
    91→    91→          "Accept-Language": "tr,en;q=0.8",
    92→    92→        },
    93→    93→        signal: AbortSignal.timeout(15000),
    94→    94→        cache: "no-store",
    95→    95→      }
    96→    96→    );
    97→    97→    if (!res.ok) return [];
    98→    98→    const html = await res.text();
    99→    99→
   100→   100→    const anchorRe = /<a\s([^>]*class="[^"]*result__a[^"]*"[^>]*)>([\s\S]*?)<\/a>/g;
   101→   101→    const snippetRe = /<a\s([^>]*class="[^"]*result__snippet[^"]*"[^>]*)>([\s\S]*?)<\/a>/g;
   102→   102→
   103→   103→    const snippets: string[] = [];
   104→   104→    let sm: RegExpExecArray | null;
   105→   105→    while ((sm = snippetRe.exec(html)) !== null) {
   106→   106→      snippets.push(tagText(sm[2]).slice(0, 600));
   107→   107→    }
   108→   108→
   109→   109→    const items: RawSearchItem[] = [];
   110→   110→    let m: RegExpExecArray | null;
   111→   111→    while ((m = anchorRe.exec(html)) !== null && items.length < num) {
   112→   112→      const hrefMatch = m[1].match(/href="([^"]+)"/);
   113→   113→      if (!hrefMatch) continue;
   114→   114→      let href = hrefMatch[1].replace(/&amp;/g, "&");
   115→   115→      if (href.startsWith("//")) href = `https:${href}`;
   116→   116→      if (/duckduckgo\.com\/l\//i.test(href)) {
   117→   117→        try {
   118→   118→          const real = new URL(href).searchParams.get("uddg");
   119→   119→          if (!real) continue;
   120→   120→          href = real;
   121→   121→        } catch {
   122→   122→          continue;
   123→   123→        }
   124→   124→      }
   125→   125→      if (!/^https?:\/\//i.test(href)) continue;
   126→   126→      if (/duckduckgo\.com\/y\.js/i.test(href)) continue; // reklam bağlantısı
   127→   127→      const title = tagText(m[2]);
   128→   128→      if (!title || title.length < 3) continue;
   129→   129→      let host = "";
   130→   130→      try {
   131→   131→        host = new URL(href).hostname;
   132→   132→      } catch {
   133→   133→        continue;
   134→   134→      }
   135→   135→      items.push({
   136→   136→        url: href,
   137→   137→        name: title,
   138→   138→        snippet: snippets[items.length] ?? "",
   139→   139→        host_name: host,
   140→   140→        date: "",
   141→   141→      });
   142→   142→    }
   143→   143→    return items;
   144→   144→  } catch (err) {
   145→   145→    console.error(`[searchWebDuckDuckGo] Arama hatası (${query}):`, err);
   146→   146→    return [];
   147→   147→  }
   148→   148→}

   181→   181→export async function translateAndClassifyBatch(
   182→   182→  items: TranslateInput[],
   183→   183→  lang: ScanLanguage
   184→   184→): Promise<TranslatedFinding[]> {
   185→   185→  if (items.length === 0) return [];
   186→   186→
   187→   187→  const userPrompt = `Kaynak dil: ${lang.name} (${lang.code}).
   188→   188→Aşağıdaki ${items.length} adet adli bilimler / olay yeri inceleme haberini Türkçe'ye çevir ve analiz et.
   189→   189→Dikkat: Reklam, genel haber, alakasız içerik varsa category="genel", importance=1 ver.
   190→   190→
   191→   191→GIRDİ (JSON):
   192→   192→${JSON.stringify(items, null, 1)}`;
   193→   193→
   194→   194→  for (let attempt = 1; attempt <= 2; attempt++) {
   195→   195→    try {
   196→   196→      const zai = await getZAI();
   197→   197→      const completion = await zai.chat.completions.create({
   198→   198→        messages: [
   199→   199→          { role: "assistant", content: TRANSLATE_SYSTEM_PROMPT },
   200→   200→          { role: "user", content: userPrompt },
   201→   201→        ],
   202→   202→        thinking: { type: "disabled" },
   203→   203→      });
   204→   204→
   205→   205→      const raw = completion?.choices?.[0]?.message?.content ?? "";
   206→   206→      const parsed = extractJsonArray(raw);
   207→   207→      if (!parsed) throw new Error("JSON ayrıştırılamadı");
   208→   208→
   209→   209→      const valid = parsed
   210→   210→        .filter((it: any) => typeof it?.index === "number" && it?.trTitle)
   211→   211→        .map((it: any) => ({
   212→   212→          index: it.index,
   213→   213→          trTitle: String(it.trTitle).slice(0, 300),
   214→   214→          trSummary: String(it.trSummary ?? "").slice(0, 1200),
   215→   215→          category: CATEGORY_KEYS.includes(it.category) ? it.category : "genel",
   216→   216→          importance: Math.min(5, Math.max(1, Number(it.importance) || 3)),
   217→   217→          isNewTech: Boolean(it.isNewTech),
   218→   218→          keywords: Array.isArray(it.keywords)
   219→   219→            ? it.keywords.slice(0, 5).map((k: any) => String(k).slice(0, 40))
   220→   220→            : [],
   221→   221→        })) as TranslatedFinding[];
   222→   222→
   223→   223→      if (valid.length > 0) return valid;
   224→   224→      throw new Error("Geçerli sonuç boş");
   225→   225→    } catch (err) {
   226→   226→      console.error(`[translateBatch] Deneme ${attempt} başarısız:`, err);
   227→   227→      if (attempt === 2) {
   228→   228→        // Fallback: çevrilmemiş ama kayıtlı olsun
   229→   229→        return items.map((it) => ({
   230→   230→          index: it.index,
   231→   231→          trTitle: it.title.slice(0, 300),
   232→   232→          trSummary: it.snippet.slice(0, 500) + " [Otomatik çeviri başarısız oldu]",
   233→   233→          category: "genel",
   234→   234→          importance: 2,
   235→   235→          isNewTech: false,
   236→   236→          keywords: [],
   237→   237→        }));
   238→   238→      }
   239→   239→      await new Promise((r) => setTimeout(r, 1200));

   245→   245→/** Metin içinden JSON dizisini çıkarır (kod bloğu / ek met toleranslı) */
   246→   246→function extractJsonArray(text: string): any[] | null {
   247→   247→  if (!text) return null;
   248→   248→  const cleaned = text.replace(/```json/gi, "```").trim();
   249→   249→  const start = cleaned.indexOf("[");
   250→   250→  const end = cleaned.lastIndexOf("]");
   251→   251→  if (start === -1 || end === -1 || end <= start) return null;
   252→   252→  try {
   253→   253→    return JSON.parse(cleaned.slice(start, end + 1));
   254→   254→  } catch {
   255→   255→    return null;
   256→   256→  }
   257→   257→}

   395→   395→    .replace(/<script[\s\S]*?<\/script>/gi, " ")
   396→   396→    .replace(/<style[\s\S]*?<\/style>/gi, " ")
   397→   397→    .replace(/<noscript[\s\S]*?<\/noscript>/gi, " ")
   398→   398→    .replace(/<(nav|header|footer|aside|form|button|svg)[\s\S]*?<\/\1>/gi, " ");
   399→   399→
   400→   400→  const paras: string[] = [];
   401→   401→  const pRe = /<p\s[^>]*>([\s\S]*?)<\/p\s*>/gi;
   402→   402→  let m: RegExpExecArray | null;
   403→   403→  let total = 0;
   404→   404→  while ((m = pRe.exec(cleaned)) !== null) {
   405→   405→    const text = tagText(m[1]);
   406→   406→    // Menü/tanıtım cümlecikleri ve tek kelimelik başlıkları ele
   407→   407→    if (text.length < 45 || text.split(/\s+/).length < 7) continue;
   408→   408→    // Aşırı bağlantı dolu paragraflar genelde navigasyondur
   409→   409→    const linkCount = (m[1].match(/<a\s/gi) ?? []).length;
   410→   410→    if (linkCount > 4) continue;
   411→   411→    paras.push(text);
   412→   412→    total += text.length;
   413→   413→    if (total >= maxChars) break;
   414→   414→  }
   415→   415→
   416→   416→  // <p> yoksa (bazı sayfalarda metin div'lerde) tüm sayfa metnine düş
   417→   417→  if (paras.length === 0) {
   418→   418→    const all = stripHtml(cleaned);
   419→   419→    if (all.length > 200) {
   420→   420→      // Cümle sınırlarına göre kaba paragraf bölümlemesi
   421→   421→      const sentences = all.split(/(?<=[.!?。!?])\s+/);
   422→   422→      let buf = "";
   423→   423→      for (const s of sentences) {
   424→   424→        buf = buf ? `${buf} ${s}` : s;
   425→   425→        if (buf.length >= 320) {
   426→   426→          paras.push(buf.trim());
   427→   427→          buf = "";
   428→   428→          if (paras.join(" ").length >= maxChars) break;
   429→   429→        }
   430→   430→      }
   431→   431→      if (buf.trim().length > 60) paras.push(buf.trim());
   432→   432→    }
   433→   433→  }
   434→   434→
   435→   435→  return paras.slice(0, 150);
   436→   436→}

   465→   465→function splitIntoChunks(paras: string[]): string[][] {
   466→   466→  const chunks: string[][] = [];
   467→   467→  let buf: string[] = [];
   468→   468→  let size = 0;
   469→   469→  for (const p of paras) {
   470→   470→    if (size + p.length > CHUNK_LIMIT && buf.length > 0) {
   471→   471→      chunks.push(buf);
   472→   472→      buf = [];
   473→   473→      size = 0;
   474→   474→    }
   475→   475→    buf.push(p);
   476→   476→    size += p.length;
   477→   477→  }
   478→   478→  if (buf.length > 0) chunks.push(buf);
   479→   479→  return chunks;
   480→   480→}

   482→   482→/** Bir parçayı (paragraf listesini) Türkçeye çevirir; paragraf sayısını korumaya çalışır */
   483→   483→async function translateChunk(chunk: string[], langName: string): Promise<string[]> {
   484→   484→  const prompt = `Aşağıda ${langName} dilinde bir haber makalesinden ${chunk.length} adet paragraf var.
   485→   485→Her paragrafı profesyonel, akıcı TÜRKÇE'ye çevir (adli bilimler / olay yeri inceleme alanı).
   486→   486→
   487→   487→KURALLAR:
   488→   488→- Girdideki paragraf SAYISINI ve SIRASINI aynen koru; paragrafları boş satırla ayır.
   489→   489→- Başlık veya yorum EKLEME; yalnızca çeviriyi üret.
   490→   490→- Teknik terimlerin Türkçe karşılığını ver, gerekirse parantezle orijinalini ekle.
   491→   491→- Bir paragraf anlamsız/geçersizse onu kısa bir "(çevrilemedi)" ibaresiyle boş geçme, atla.
   492→   492→
   493→   493→PARAGRAFLAR:
   494→   494→${chunk.map((p, i) => `--${i + 1}--\n${p}`).join("\n\n")}`;
   495→   495→
   496→   496→  for (let attempt = 1; attempt <= 2; attempt++) {
   497→   497→    try {
   498→   498→      const zai = await getZAI();
   499→   499→      const completion = await zai.chat.completions.create({
   500→   500→        messages: [
   501→   501→          {
   502→   502→            role: "assistant",
   503→   503→            content:
   504→   504→              "Sen adli bilimler alanında uzmanlaşmış profesyonel bir çevirmensin. Yalnızca istenen çeviriyi, paragraf yapısını koruyarak üretirsin.",
   505→   505→          },
   506→   506→          { role: "user", content: prompt },
   507→   507→        ],
   508→   508→        thinking: { type: "disabled" },
   509→   509→      });
   510→   510→      const raw = completion?.choices?.[0]?.message?.content ?? "";
   511→   511→      const parts = raw
   512→   512→        .replace(/```[a-z]*\n?/gi, "")
   513→   513→        .split(/\n{2,}/)
   514→   514→        .map((s) => s.replace(/^--\d+--\s*/g, "").trim())
   515→   515→        .filter((s) => s.length > 2);
   516→   516→      if (parts.length >= Math.max(1, Math.floor(chunk.length * 0.6))) return parts;
   517→   517→      throw new Error(`Paragraf sayısı uyuşmadı (${parts.length}/${chunk.length})`);
   518→   518→    } catch (err) {
   519→   519→      console.error(`[translateChunk] Deneme ${attempt} başarısız:`, err);
   520→   520→      if (attempt === 2) {
   521→   521→        // Çevrilemeyen parçanın orijinali kalsın (kullanıcı yine de okuyabilsin)
   522→   522→        return chunk.map((p) => `${p} *(çevrilemedi)*`);
   523→   523→      }
   524→   524→      await new Promise((r) => setTimeout(r, 1000));
   525→   525→    }
   526→   526→  }
   527→   527→  return chunk;
   528→   528→}

   534→   534→export async function fetchAndTranslateFullArticle(
   535→   535→  url: string,
   536→   536→  originalTitle: string,
   537→   537→  trTitle: string,
   538→   538→  language: string
   539→   539→): Promise<FullArticleResult> {
   540→   540→  const { html, engine } = await fetchPageHtmlForText(url);
   541→   541→
   542→   542→  if (engine === "none" || !html) {
   543→   543→    // Sayfa okunamadı — boş sonuç; çağıran taraf özete düşer
   544→   544→    return { title: trTitle, paragraphs: [], wordCount: 0, engine: "none", translated: false };
   545→   545→  }

   558→   558→  }

   562→   562→  }
   563→   563→
   564→   564→  const chunks = splitIntoChunks(sourceParas);

   575→   575→    wordCount: paragraphs.join(" ").split(/\s+/).filter(Boolean).length,

   578→   578→  };
   579→   579→}

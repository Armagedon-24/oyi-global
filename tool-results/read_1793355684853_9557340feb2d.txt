    18→export async function getZAI(): Promise<ZAI> {
    19→  if (zaiInstance) return zaiInstance;
    20→  if (!zaiInitPromise) {
    21→    zaiInitPromise = (async () => {
    22→      const baseUrl = process.env.ZAI_BASE_URL?.trim();
    23→      const apiKey = process.env.ZAI_API_KEY?.trim();
    24→      if (baseUrl && apiKey) {
    25→        // SDK kurucusu tip tanımında private olarak işaretlidir; ortam
    26→        // değişkenleriyle doğrudan yapılandırma için güvenli tip geçişi gerekir.
    27→        const Ctor = ZAI as unknown as new (config: Record<string, string>) => ZAI;
    28→        zaiInstance = new Ctor({
    29→          baseUrl,
    30→          apiKey,
    31→          chatId: process.env.ZAI_CHAT_ID?.trim() ?? "",
    32→          userId: process.env.ZAI_USER_ID?.trim() ?? "",
    33→          token: process.env.ZAI_TOKEN?.trim() ?? "",
    34→        });
    35→        return zaiInstance;
    36→      }
    37→      zaiInstance = await ZAI.create();
    38→      return zaiInstance;
    39→    })();
    40→    zaiInitPromise.catch(() => {
    41→      // Başarısız init sonrası tekrar denemeye izin ver
    42→      zaiInitPromise = null;
    43→    });
    44→  }
    45→  return zaiInitPromise;
    46→}

    81→/**
    82→ * DuckDuckGo HTML araması (API anahtarı gerektirmez). Z-AI web_search
    83→ * kullanılamadığında yedek arama motoru olarak çalışır.

    85→export async function searchWebDuckDuckGo(query: string, num = 10): Promise<RawSearchItem[]> {
    86→  try {
    87→    const res = await fetch(
    88→      `https://html.duckduckgo.com/html/?q=${encodeURIComponent(query)}`,
    89→      {
    90→        headers: {
    91→          "User-Agent": DDG_UA,
    92→          Accept: "text/html,application/xhtml+xml",
    93→          "Accept-Language": "tr,en;q=0.8",
    94→        },
    95→        signal: AbortSignal.timeout(15000),
    96→        cache: "no-store",
    97→      }
    98→    );
    99→    if (!res.ok) return [];
   100→    const html = await res.text();
   101→
   102→    const anchorRe = /<a\s([^>]*class="[^"]*result__a[^"]*"[^>]*)>([\s\S]*?)<\/a>/g;
   103→    const snippetRe = /<a\s([^>]*class="[^"]*result__snippet[^"]*"[^>]*)>([\s\S]*?)<\/a>/g;
   104→
   105→    const snippets: string[] = [];
   106→    let sm: RegExpExecArray | null;
   107→    while ((sm = snippetRe.exec(html)) !== null) {
   108→      snippets.push(tagText(sm[2]).slice(0, 600));
   109→    }
   110→
   111→    const items: RawSearchItem[] = [];
   112→    let m: RegExpExecArray | null;
   113→    while ((m = anchorRe.exec(html)) !== null && items.length < num) {
   114→      const hrefMatch = m[1].match(/href="([^"]+)"/);
   115→      if (!hrefMatch) continue;
   116→      let href = hrefMatch[1].replace(/&amp;/g, "&");
   117→      if (href.startsWith("//")) href = `https:${href}`;
   118→      if (/duckduckgo\.com\/l\//i.test(href)) {
   119→        try {
   120→          const real = new URL(href).searchParams.get("uddg");
   121→          if (!real) continue;
   122→          href = real;
   123→        } catch {
   124→          continue;
   125→        }
   126→      }
   127→      if (!/^https?:\/\//i.test(href)) continue;
   128→      if (/duckduckgo\.com\/y\.js/i.test(href)) continue; // reklam bağlantısı
   129→      const title = tagText(m[2]);
   130→      if (!title || title.length < 3) continue;
   131→      let host = "";
   132→      try {
   133→        host = new URL(href).hostname;
   134→      } catch {
   135→        continue;
   136→      }
   137→      items.push({
   138→        url: href,
   139→        name: title,
   140→        snippet: snippets[items.length] ?? "",
   141→        host_name: host,
   142→        date: "",
   143→      });
   144→    }
   145→    return items;
   146→  } catch (err) {
   147→    console.error(`[searchWebDuckDuckGo] Arama hatası (${query}):`, err);
   148→    return [];
   149→  }
   150→}

   157→/** Metni cümle/sözcük sınırından taşmadan parçalara böler */
   158→function splitForHttpTranslate(text: string, maxLen: number): string[] {
   159→  if (text.length <= maxLen) return [text];
   160→  const parts: string[] = [];
   161→  let rest = text;
   162→  while (rest.length > maxLen) {
   163→    const window = rest.slice(0, maxLen);
   164→    let cut = -1;
   165→    for (const sep of [". ", "! ", "? ", "; ", ", ", " "]) {
   166→      const idx = window.lastIndexOf(sep);
   167→      if (idx > maxLen * 0.5) {
   168→        cut = idx + sep.length;
   169→        break;
   170→      }
   171→    }
   172→    if (cut <= 0) cut = maxLen;
   173→    parts.push(rest.slice(0, cut));
   174→    rest = rest.slice(cut);
   175→  }
   176→  if (rest.trim()) parts.push(rest);
   177→  return parts;
   178→}

   180→async function translateViaClients5(text: string, src: string): Promise<string> {
   181→  const res = await fetch(
   182→    `https://clients5.google.com/translate_a/t?client=dict-chrome-ex&sl=${encodeURIComponent(src)}&tl=tr&q=${encodeURIComponent(text)}`,
   183→    { headers: { "User-Agent": DDG_UA }, signal: AbortSignal.timeout(9000), cache: "no-store" }
   184→  );
   185→  if (!res.ok) return "";
   186→  const data: unknown = await res.json();
   187→  if (Array.isArray(data) && Array.isArray(data[0])) {
   188→    return (data[0] as unknown[])
   189→      .map((seg) => (Array.isArray(seg) ? String(seg[0] ?? "") : String(seg ?? "")))
   190→      .join("")
   191→      .trim();
   192→  }
   193→  if (Array.isArray(data) && typeof data[0] === "string") return (data[0] as string).trim();
   194→  return "";
   195→}

   197→async function translateViaGtx(text: string, src: string): Promise<string> {
   198→  const res = await fetch(
   199→    `https://translate.googleapis.com/translate_a/single?client=gtx&sl=${encodeURIComponent(src)}&tl=tr&dt=t&q=${encodeURIComponent(text)}`,
   200→    { headers: { "User-Agent": DDG_UA }, signal: AbortSignal.timeout(12000), cache: "no-store" }
   201→  );
   202→  if (!res.ok) return "";
   203→  const data: unknown = await res.json();
   204→  if (Array.isArray(data) && Array.isArray(data[0])) {
   205→    return (data[0] as unknown[]).map((seg) => String((seg as unknown[])?.[0] ?? "")).join("").trim();
   206→  }
   207→  return "";
   208→}

   210→async function translateViaMyMemory(text: string, src: string): Promise<string> {
   211→  const res = await fetch(
   212→    `https://api.mymemory.translated.net/get?q=${encodeURIComponent(text.slice(0, 450))}&langpair=${encodeURIComponent(src)}|tr`,
   213→    { headers: { "User-Agent": DDG_UA }, signal: AbortSignal.timeout(9000), cache: "no-store" }
   214→  );
   215→  if (!res.ok) return "";
   216→  const data: any = await res.json();
   217→  const t = data?.responseData?.translatedText;
   218→  return typeof t === "string" && !/^MYMEMORY WARNING/i.test(t) ? t.trim() : "";
   219→}

   221→/**
   222→ * Z-AI kullanılamadığında (örn. Vercel) ücretsiz servislerle Türkçeye çevirir.
   223→ * Hepsinin başarısız olduğu durumda boş metin döner; çağıran taraf orijinali kullanır.
   224→ */
   225→export async function httpTranslate(text: string, src = "auto"): Promise<string> {
   226→  const clean = (text ?? "").trim();
   227→  if (!clean) return "";
   228→  const source = src && src.trim().length >= 2 ? src.trim() : "auto";
   229→  const out: string[] = [];
   230→  for (const part of splitForHttpTranslate(clean, 1200)) {
   231→    try {
   232→      const t =
   233→        (await translateViaClients5(part, source).catch(() => "")) ||
   234→        (await translateViaGtx(part, source).catch(() => "")) ||
   235→        (await translateViaMyMemory(part, source).catch(() => ""));
   236→      if (!t) return "";
   237→      out.push(t);
   238→    } catch {
   239→      return "";
   240→    }
   241→  }
   242→  return out.join(" ").trim();
   243→}

   245→/**
   246→ * Google News RSS araması (anahtar gerektirmez, bot engellemesine dayanıklı).
   247→ * DuckDuckGo da başarısız olursa son yedek arama kaynağı olarak kullanılır.
   248→ * Google News bağlantıları kodlanmış yönlendirme adresleri olduğu için
   249→ * gerçek makale URL'si base64 yükünden çıkarılır; çözülemeyenler atlanır.
   250→ */
   251→export async function searchWebGoogleNewsRss(query: string, num = 10): Promise<RawSearchItem[]> {
   252→  try {
   253→    const res = await fetch(
   254→      `https://news.google.com/rss/search?q=${encodeURIComponent(query)}`,
   255→      {
   256→        headers: { "User-Agent": DDG_UA, Accept: "application/rss+xml,application/xml,text/xml" },
   257→        signal: AbortSignal.timeout(15000),
   258→        cache: "no-store",
   259→      }
   260→    );
   261→    if (!res.ok) return [];
   262→    const xml = await res.text();
   263→
   264→    const items: RawSearchItem[] = [];
   265→    const itemRe = /<item>([\s\S]*?)<\/item>/g;
   266→    let m: RegExpExecArray | null;
   267→    while ((m = itemRe.exec(xml)) !== null && items.length < num) {
   268→      const block = m[1];
   269→      const pickTag = (tag: string): string => {
   270→        const t = block.match(new RegExp(`<${tag}[^>]*>([\\s\\S]*?)<\\/${tag}>`, "i"));
   271→        return t ? t[1].replace(/<!\[CDATA\[/g, "").replace(/\]\]>/g, "").trim() : "";
   272→      };
   273→      const title = tagText(pickTag("title"))
   274→        .replace(/\s+-\s+[^-]{2,60}$/u, "")
   275→        .trim();
   276→      const rawLink = pickTag("link");
   277→      const link = decodeGoogleNewsLink(rawLink);
   278→      if (!link || !title || title.length < 8) continue;
   279→      const pubDate = pickTag("pubDate");
   280→      let date = "";
   281→      if (pubDate) {
   282→        const d = new Date(pubDate);
   283→        if (!isNaN(d.getTime())) date = d.toISOString().slice(0, 10);
   284→      }
   285→      let host = "";
   286→      try {
   287→        host = new URL(link).hostname;
   288→      } catch {
   289→        continue;
   290→      }
   291→      items.push({ url: link, name: title, snippet: "", host_name: host, date });
   292→    }
   293→    return items;
   294→  } catch (err) {
   295→    console.error(`[searchWebGoogleNewsRss] Arama hatası (${query}):`, err);
   296→    return [];
   297→  }
   298→}

   300→/** Google News yönlendirme bağlantısını gerçek makale adresine çözer */
   301→function decodeGoogleNewsLink(link: string): string {
   302→  try {
   303→    const u = new URL(link);
   304→    if (!/news\.google\.com$/i.test(u.hostname)) return link;
   305→    const id = u.pathname.split("/articles/")[1] ?? "";
   306→    if (!id) return "";
   307→    const decoded = Buffer.from(id.replace(/-/g, "+").replace(/_/g, "/"), "base64").toString("binary");
   308→    const m = decoded.match(/https?:\/\/[!-~]+/);
   309→    return m ? m[0] : "";
   310→  } catch {
   311→    return "";
   312→  }
   313→}

   345→/** LLM ile toplu çeviri + kategorizasyon. Hata durumunda kısmi sonuç döner. */
   346→export async function translateAndClassifyBatch(
   347→  items: TranslateInput[],
   348→  lang: ScanLanguage
   349→): Promise<TranslatedFinding[]> {
   350→  if (items.length === 0) return [];
   351→
   352→  const userPrompt = `Kaynak dil: ${lang.name} (${lang.code}).
   353→Aşağıdaki ${items.length} adet adli bilimler / olay yeri inceleme haberini Türkçe'ye çevir ve analiz et.
   354→Dikkat: Reklam, genel haber, alakasız içerik varsa category="genel", importance=1 ver.
   355→
   356→GIRDİ (JSON):
   357→${JSON.stringify(items, null, 1)}`;
   358→
   359→  for (let attempt = 1; attempt <= 2; attempt++) {
   360→    try {
   361→      const zai = await getZAI();
   362→      const completion = await zai.chat.completions.create({
   363→        messages: [
   364→          { role: "assistant", content: TRANSLATE_SYSTEM_PROMPT },
   365→          { role: "user", content: userPrompt },
   366→        ],
   367→        thinking: { type: "disabled" },
   368→      });
   369→
   370→      const raw = completion?.choices?.[0]?.message?.content ?? "";
   371→      const parsed = extractJsonArray(raw);
   372→      if (!parsed) throw new Error("JSON ayrıştırılamadı");
   373→
   374→      const valid = parsed
   375→        .filter((it: any) => typeof it?.index === "number" && it?.trTitle)
   376→        .map((it: any) => ({
   377→          index: it.index,
   378→          trTitle: String(it.trTitle).slice(0, 300),
   379→          trSummary: String(it.trSummary ?? "").slice(0, 1200),
   380→          category: CATEGORY_KEYS.includes(it.category) ? it.category : "genel",
   381→          importance: Math.min(5, Math.max(1, Number(it.importance) || 3)),
   382→          isNewTech: Boolean(it.isNewTech),
   383→          keywords: Array.isArray(it.keywords)
   384→            ? it.keywords.slice(0, 5).map((k: any) => String(k).slice(0, 40))
   385→            : [],
   386→        })) as TranslatedFinding[];
   387→
   388→      if (valid.length > 0) return valid;
   389→      throw new Error("Geçerli sonuç boş");
   390→    } catch (err) {
   391→      console.error(`[translateBatch] Deneme ${attempt} başarısız:`, err);
   392→      if (attempt === 2) {
   393→        // Fallback: Z-AI yoksa ücretsiz HTTP çevirisi; o da olmazsa orijinal kaydedilir
   394→        const source = lang?.code || "auto";
   395→        return Promise.all(
   396→          items.map(async (it) => {
   397→            const trTitle = await httpTranslate(it.title.slice(0, 300), source);
   398→            const trSummary = await httpTranslate(it.snippet.slice(0, 700), source);
   399→            return {
   400→              index: it.index,
   401→              trTitle: (trTitle || it.title).slice(0, 300),
   402→              trSummary:
   403→                (trSummary || it.snippet.slice(0, 500)) +
   404→                (trTitle || trSummary ? "" : " [Otomatik çeviri başarısız oldu]"),
   405→              category: "genel",
   406→              importance: 2,
   407→              isNewTech: false,
   408→              keywords: [],
   409→            };
   410→          })
   411→        );
   412→      }
   413→      await new Promise((r) => setTimeout(r, 1200));
   414→    }
   415→  }
   416→  return [];
   417→}

   419→/** Metin içinden JSON dizisini çıkarır (kod bloğu / ek met toleranslı) */
   420→function extractJsonArray(text: string): any[] | null {
   421→  if (!text) return null;
   422→  const cleaned = text.replace(/```json/gi, "```").trim();
   423→  const start = cleaned.indexOf("[");
   424→  const end = cleaned.lastIndexOf("]");
   425→  if (start === -1 || end === -1 || end <= start) return null;
   426→  try {
   427→    return JSON.parse(cleaned.slice(start, end + 1));
   428→  } catch {
   429→    return null;
   430→  }
   431→}

   542→const FULL_TEXT_LIMIT = 12000; // Kaynak metin üst sınırı (karakter)
   543→const CHUNK_LIMIT = 3800; // Her çeviri parçasının üst sınırı

   545→/** HTML tag dizesinin iç metnini temizler */
   546→function tagText(html: string): string {
   547→  return html
   548→    .replace(/<script[\s\S]*?<\/script>/gi, " ")
   549→    .replace(/<style[\s\S]*?<\/style>/gi, " ")
   550→    .replace(/<[^>]+>/g, " ")
   551→    .replace(/&nbsp;/gi, " ")
   552→    .replace(/&amp;/gi, "&")
   553→    .replace(/&lt;/gi, "<")
   554→    .replace(/&gt;/gi, ">")
   555→    .replace(/&quot;/gi, '"')
   556→    .replace(/&#0?39;/gi, "'")
   557→    .replace(/&#x27;/gi, "'")
   558→    .replace(/\s+/g, " ")
   559→    .trim();
   560→}

   562→/**
   563→ * Makale HTML'inden paragraf yapısını koruyarak ana metni çıkarır:
   564→ * script/style/nav/header/footer/aside/form atılır, <p> blokları sırayla toplanır.
   565→ */
   566→export function extractArticleParagraphs(html: string, maxChars = FULL_TEXT_LIMIT): string[] {
   567→  if (!html) return [];
   568→  const cleaned = html
   569→    .replace(/<script[\s\S]*?<\/script>/gi, " ")
   570→    .replace(/<style[\s\S]*?<\/style>/gi, " ")
   571→    .replace(/<noscript[\s\S]*?<\/noscript>/gi, " ")
   572→    .replace(/<(nav|header|footer|aside|form|button|svg)[\s\S]*?<\/\1>/gi, " ");
   573→
   574→  const paras: string[] = [];
   575→  const pRe = /<p\s[^>]*>([\s\S]*?)<\/p\s*>/gi;
   576→  let m: RegExpExecArray | null;
   577→  let total = 0;
   578→  while ((m = pRe.exec(cleaned)) !== null) {
   579→    const text = tagText(m[1]);
   580→    // Menü/tanıtım cümlecikleri ve tek kelimelik başlıkları ele
   581→    if (text.length < 45 || text.split(/\s+/).length < 7) continue;
   582→    // Aşırı bağlantı dolu paragraflar genelde navigasyondur
   583→    const linkCount = (m[1].match(/<a\s/gi) ?? []).length;
   584→    if (linkCount > 4) continue;
   585→    paras.push(text);
   586→    total += text.length;
   587→    if (total >= maxChars) break;
   588→  }
   589→
   590→  // <p> yoksa (bazı sayfalarda metin div'lerde) tüm sayfa metnine düş
   591→  if (paras.length === 0) {
   592→    const all = stripHtml(cleaned);
   593→    if (all.length > 200) {
   594→      // Cümle sınırlarına göre kaba paragraf bölümlemesi
   595→      const sentences = all.split(/(?<=[.!?。!?])\s+/);
   596→      let buf = "";
   597→      for (const s of sentences) {
   598→        buf = buf ? `${buf} ${s}` : s;
   599→        if (buf.length >= 320) {
   600→          paras.push(buf.trim());
   601→          buf = "";
   602→          if (paras.join(" ").length >= maxChars) break;
   603→        }
   604→      }
   605→      if (buf.trim().length > 60) paras.push(buf.trim());
   606→    }
   607→  }
   608→
   609→  return paras.slice(0, 150);
   610→}

   615→async function fetchPageHtmlForText(
   616→  url: string
   617→): Promise<{ html: string; engine: FullArticleResult["engine"] }> {
   618→  try {
   619→    const { fetchPageHtmlDirect } = await import("./image-extract");
   620→    const html = await fetchPageHtmlDirect(url, 11000);
   621→    if (html && html.length > 500) return { html, engine: "direct" };
   622→  } catch {
   623→    /* yedeğe düş */
   624→  }

   628→      zai.functions.invoke("page_reader", { url }) as Promise<any>,
   629→      new Promise((_, rej) => setTimeout(() => rej(new Error("zaman aşımı")), 22000)),
   630→    ])) as any;
   631→    const html: string = page?.data?.html ?? "";
   632→    if (html) return { html, engine: "page_reader" };
   633→  } catch {
   634→    /* her iki yol da başarısız */
   635→  }

   639→function splitIntoChunks(paras: string[]): string[][] {
   640→  const chunks: string[][] = [];
   641→  let buf: string[] = [];
   642→  let size = 0;
   643→  for (const p of paras) {
   644→    if (size + p.length > CHUNK_LIMIT && buf.length > 0) {
   645→      chunks.push(buf);
   646→      buf = [];
   647→      size = 0;
   648→    }
   649→    buf.push(p);
   650→    size += p.length;
   651→  }
   652→  if (buf.length > 0) chunks.push(buf);
   653→  return chunks;
   654→}

   656→/** Bir parçayı (paragraf listesini) Türkçeye çevirir; paragraf sayısını korumaya çalışır */
   657→async function translateChunk(chunk: string[], langName: string): Promise<string[]> {
   658→  const prompt = `Aşağıda ${langName} dilinde bir haber makalesinden ${chunk.length} adet paragraf var.
   659→Her paragrafı profesyonel, akıcı TÜRKÇE'ye çevir (adli bilimler / olay yeri inceleme alanı).
   660→
   661→KURALLAR:
   662→- Girdideki paragraf SAYISINI ve SIRASINI aynen koru; paragrafları boş satırla ayır.
   663→- Başlık veya yorum EKLEME; yalnızca çeviriyi üret.
   664→- Teknik terimlerin Türkçe karşılığını ver, gerekirse parantezle orijinalini ekle.
   665→- Bir paragraf anlamsız/geçersizse onu kısa bir "(çevrilemedi)" ibaresiyle boş geçme, atla.
   666→
   667→PARAGRAFLAR:
   668→${chunk.map((p, i) => `--${i + 1}--\n${p}`).join("\n\n")}`;
   669→
   670→  for (let attempt = 1; attempt <= 2; attempt++) {
   671→    try {
   672→      const zai = await getZAI();
   673→      const completion = await zai.chat.completions.create({
   674→        messages: [
   675→          {
   676→            role: "assistant",
   677→            content:
   678→              "Sen adli bilimler alanında uzmanlaşmış profesyonel bir çevirmensin. Yalnızca istenen çeviriyi, paragraf yapısını koruyarak üretirsin.",
   679→          },
   680→          { role: "user", content: prompt },
   681→        ],
   682→        thinking: { type: "disabled" },
   683→      });
   684→      const raw = completion?.choices?.[0]?.message?.content ?? "";
   685→      const parts = raw
   686→        .replace(/```[a-z]*\n?/gi, "")
   687→        .split(/\n{2,}/)
   688→        .map((s) => s.replace(/^--\d+--\s*/g, "").trim())
   689→        .filter((s) => s.length > 2);
   690→      if (parts.length >= Math.max(1, Math.floor(chunk.length * 0.6))) return parts;
   691→      throw new Error(`Paragraf sayısı uyuşmadı (${parts.length}/${chunk.length})`);
   692→    } catch (err) {
   693→      console.error(`[translateChunk] Deneme ${attempt} başarısız:`, err);
   694→      if (attempt === 2) {
   695→        // Z-AI yoksa ücretsiz HTTP çevirisi; çevrilemeyen paragraf orijinal kalsın
   696→        return Promise.all(
   697→          chunk.slice(0, 40).map(async (p) => {
   698→            const t = p.length <= 1200 ? await httpTranslate(p, langName).catch(() => "") : "";
   699→            return t || `${p} *(çevrilemedi)*`;
   700→          })
   701→        );
   702→      }
   703→      await new Promise((r) => setTimeout(r, 1000));
   704→    }
   705→  }
   706→  return chunk;
   707→}

   709→/**
   710→ * Tam Oku: makalenin TAM metnini kaynaktan okur ve Türkçeye çevirir.
   711→ * Kaynak dil Türkçeyse çeviri atlanır. Hata durumunda mümkün olan kısmı döndürür.
   712→ */
   713→export async function fetchAndTranslateFullArticle(
   714→  url: string,
   715→  originalTitle: string,
   716→  trTitle: string,
   717→  language: string
   718→): Promise<FullArticleResult> {
   719→  const { html, engine } = await fetchPageHtmlForText(url);
   720→
   721→  if (engine === "none" || !html) {
   722→    // Sayfa okunamadı — boş sonuç; çağıran taraf özete düşer
   723→    return { title: trTitle, paragraphs: [], wordCount: 0, engine: "none", translated: false };
   724→  }
   725→
   726→  const sourceParas = extractArticleParagraphs(html);
   727→  const isTurkish = /^tr\b/i.test(language) || /türk/i.test(language);
   728→
   729→  if (isTurkish) {
   730→    return {
   731→      title: trTitle,
   732→      paragraphs: sourceParas,
   733→      wordCount: sourceParas.join(" ").split(/\s+/).filter(Boolean).length,
   734→      engine,
   735→      translated: false,
   736→    };
   737→  }
   738→
   739→  if (sourceParas.length === 0) {
   740→    return { title: trTitle, paragraphs: [], wordCount: 0, engine, translated: false };
   741→  }
   742→
   743→  const chunks = splitIntoChunks(sourceParas);
   744→  const translatedParts: string[] = [];
   745→  for (const chunk of chunks) {
   746→    const done = await translateChunk(chunk, language);
   747→    translatedParts.push(...done);
   748→  }
   749→
   750→  const paragraphs = translatedParts.filter((p) => p && p.trim()).slice(0, 150);
   751→  return {
   752→    title: trTitle,
   753→    paragraphs,
   754→    wordCount: paragraphs.join(" ").split(/\s+/).filter(Boolean).length,
   755→    engine,
   756→    translated: true,
   757→  };
   758→}

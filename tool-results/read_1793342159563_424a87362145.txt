    31→/** Köşeden sürükleme durumu — sayfa parmağı izler */
    32→interface DragState {
    33→  dir: "next" | "prev";
    34→  from: number;
    35→  to: number;
    36→  p: number; // 0..1 çevirme ilerlemesi
    37→  settling: "commit" | "back" | null;
    38→}
    39→
    40→/** Sürükleme ilerlemesine göre yaprağın bükülmüş (esneyen) transform'u */
    41→function leafTransform(ds: { dir: "next" | "prev"; p: number }): string {
    42→  const bend = Math.sin(Math.PI * ds.p); // ortada en çok esner
    43→  const angle = ds.dir === "next" ? -ds.p * 180 : -(1 - ds.p) * 180;
    44→  const rz = (ds.dir === "next" ? 1 : -1) * bend * 2.8;
    45→  const tx = (ds.dir === "next" ? 1 : -1) * bend * 0.5;
    46→  const ty = -bend * 1.3;
    47→  const sx = 1 - bend * 0.045;
    48→  const sy = 1 + bend * 0.028;
    49→  return `rotateY(${angle.toFixed(2)}deg) rotateZ(${rz.toFixed(2)}deg) translate3d(${tx.toFixed(2)}%, ${ty.toFixed(2)}%, 0) scale(${sx.toFixed(3)}, ${sy.toFixed(3)})`;
    50→}
    51→
    52→/** Sürüklemede dış yarı yaprağın faz farkıyla katlanması (kâğıt kırışması) */
    53→function foldTransform(ds: { p: number }): string {
    54→  const bend = Math.sin(Math.PI * ds.p);
    55→  return `rotateY(${(-bend * 34).toFixed(2)}deg)`;
    56→}

    72→function formatDate(s: string | null): string | null {
    73→  if (!s) return null;
    74→  const d = new Date(s);
    75→  if (isNaN(d.getTime())) return null;
    76→  return d.toLocaleDateString("tr-TR", { day: "numeric", month: "long", year: "numeric" });
    77→}

    79→function parseKeywords(raw: string): string[] {
    80→  try {
    81→    const arr = JSON.parse(raw);
    82→    return Array.isArray(arr) ? arr.slice(0, 5) : [];
    83→  } catch {
    84→    return [];
    85→  }
    86→}
    87→
    88→function clip(text: string, max: number): string {
    89→  if (text.length <= max) return text;
    90→  const cut = text.slice(0, max);
    91→  return cut.replace(/\s+\S*$/, "") + "…";
    92→}

    94→/** Öne çıkan görseli döndürür (proxy adresiyle) */
    95→function coverImage(f: FindingItem) {
    96→  const imgs = parseFindingImages(f.images);
    97→  return imgs.length > 0 ? imgs[0] : null;
    98→}

   100→export function MagazineView({
   101→  findings,
   102→  loading,
   103→  scanning,
   104→  onStartScan,
   105→  onToggleBookmark,
   106→}: MagazineViewProps) {
   107→  const [page, setPage] = useState(0);
   108→  const [flip, setFlip] = useState<{ dir: "next" | "prev"; from: number; to: number } | null>(null);
   109→  const [drag, setDrag] = useState<DragState | null>(null);
   110→  const [reading, setReading] = useState<FindingItem | null>(null);
   111→  const flipTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
   112→  const touchX = useRef<number | null>(null);
   113→  const frameRef = useRef<HTMLDivElement | null>(null);
   114→  const leafRef = useRef<HTMLDivElement | null>(null);
   115→  const foldRef = useRef<HTMLDivElement | null>(null);
   116→  const dragRef = useRef<DragState | null>(null);
   117→  const dragStartX = useRef(0);
   118→  const dragWidth = useRef(640);
   119→  const dragMoved = useRef(false);
   120→  const lastDragAt = useRef(0);
   121→
   122→  // ---- Sayı bilgisi (dergi künyesi) ----
   123→  const issue = useMemo(() => {
   124→    const now = new Date();
   125→    const month = now.toLocaleDateString("tr-TR", { month: "long" }).toLocaleUpperCase("tr-TR");
   126→    const no = (now.getFullYear() - 2024) * 12 + now.getMonth() + 1;
   127→    return { month, no, year: now.getFullYear() };
   128→  }, []);
   129→
   130→  // ---- Sayfa modeli ----
   131→  const articles = useMemo(() => findings.slice(0, MAX_ARTICLES), [findings]);
   132→
   133→  const pages = useMemo<MagPage[]>(() => {
   134→    if (articles.length === 0) return [{ kind: "empty" }];
   135→    let featIdx = articles.findIndex((a) => coverImage(a) !== null);
   136→    if (featIdx < 0) featIdx = 0;
   137→    const featured = articles[featIdx];
   138→    const secondary = articles.filter((a) => a.id !== featured.id).slice(0, 3);
   139→    const tocEntries = articles.map((f, i) => ({ f, folio: i + 2 }));
   140→    return [
   141→      { kind: "cover", featured, secondary, totalCount: findings.length },
   142→      { kind: "toc", entries: tocEntries },
   143→      ...articles.map((f, i) => ({ kind: "article" as const, f, folio: i + 2, n: i + 1 })),
   144→    ];
   145→  }, [articles, findings.length]);
   146→
   147→  const totalPages = pages.length;
   148→
   149→  // Filtre değişince sayfa taşarsa başa dön
   150→  useEffect(() => {
   151→    if (page > totalPages - 1) setPage(Math.max(0, totalPages - 1));
   152→  }, [totalPages, page]);
   153→
   154→  useEffect(
   155→    () => () => {
   156→      if (flipTimer.current) clearTimeout(flipTimer.current);
   157→    },
   158→    []
   159→  );
   160→
   161→  // ---- Sayfa çevirme motoru ----
   162→  const goTo = useCallback(
   163→    (target: number, dir?: "next" | "prev") => {
   164→      if (flip || dragRef.current) return;
   165→      const clamped = Math.max(0, Math.min(totalPages - 1, target));
   166→      if (clamped === page) return;
   167→      const d: "next" | "prev" = dir ?? (clamped > page ? "next" : "prev");
   168→      setFlip({ dir: d, from: page, to: clamped });
   169→      setPage(clamped);
   170→      if (flipTimer.current) clearTimeout(flipTimer.current);
   171→      flipTimer.current = setTimeout(() => setFlip(null), FLIP_MS + 40);
   172→    },
   173→    [flip, page, totalPages]
   174→  );
   175→
   176→  const goNext = useCallback(() => goTo(page + 1, "next"), [goTo, page]);
   177→  const goPrev = useCallback(() => goTo(page - 1, "prev"), [goTo, page]);
   178→
   179→  // Klavye ile gezinme (form alanları hariç)
   180→  useEffect(() => {
   181→    const onKey = (e: KeyboardEvent) => {
   182→      const tag = (e.target as HTMLElement | null)?.tagName;
   183→      if (tag && ["INPUT", "TEXTAREA", "SELECT"].includes(tag)) return;
   184→      if (reading || flip || dragRef.current) return;
   185→      if (e.key === "ArrowRight") goNext();
   186→      else if (e.key === "ArrowLeft") goPrev();
   187→    };
   188→    window.addEventListener("keydown", onKey);
   189→    return () => window.removeEventListener("keydown", onKey);
   190→  }, [goNext, goPrev, reading, flip]);
   191→
   192→  // Dokunmatik kaydırma
   193→  const onTouchStart = (e: React.TouchEvent) => {
   194→    touchX.current = e.touches[0].clientX;
   195→  };
   196→  const onTouchEnd = (e: React.TouchEvent) => {
   197→    if (Date.now() - lastDragAt.current < 500) return; // köşe sürüklemesiyle çakışmasın
   198→    if (touchX.current === null) return;
   199→    const delta = e.changedTouches[0].clientX - touchX.current;
   200→    touchX.current = null;
   201→    if (delta < -48) goNext();
   202→    else if (delta > 48) goPrev();
   203→  };
   204→
   205→  // ---- Köşeden tutup sürükleyerek çevirme (sayfa el ile çevriliyormuş hissi) ----
   206→  const onEarPointerDown = (dir: "next" | "prev") => (e: React.PointerEvent<HTMLButtonElement>) => {
   207→    if (flip || dragRef.current || reading || loading) return;
   208→    if (dir === "next" ? page >= totalPages - 1 : page === 0) return;
   209→    e.preventDefault();
   210→    try {
   211→      e.currentTarget.setPointerCapture(e.pointerId);
   212→    } catch {
   213→      /* capture opsiyonel */
   214→    }
   215→    const frame = frameRef.current;
   216→    dragWidth.current = frame ? frame.getBoundingClientRect().width : 640;
   217→    dragStartX.current = e.clientX;
   218→    dragMoved.current = false;
   219→    lastDragAt.current = Date.now();
   220→    const ds: DragState = { dir, from: page, to: dir === "next" ? page + 1 : page - 1, p: 0, settling: null };
   221→    dragRef.current = ds;
   222→    setDrag(ds);
   223→  };
   224→
   225→  const dragging = drag !== null;
   226→  useEffect(() => {
   227→    if (!dragging) return;
   228→    let raf = 0;
   229→    let pendingX = 0;
   230→    const applyMove = () => {
   231→      raf = 0;
   232→      const ds = dragRef.current;
   233→      if (!ds || ds.settling) return;
   234→      const raw = ds.dir === "next" ? dragStartX.current - pendingX : pendingX - dragStartX.current;
   235→      if (Math.abs(raw) > 6) dragMoved.current = true;
   236→      const p = Math.max(0, Math.min(1, raw / (dragWidth.current * 0.72)));
   237→      if (p === ds.p) return;
   238→      const next = { ...ds, p };
   239→      dragRef.current = next;
   240→      setDrag(next);
   241→    };
   242→    const onMove = (e: PointerEvent) => {
   243→      pendingX = e.clientX;
   244→      if (!raf) raf = requestAnimationFrame(applyMove);
   245→    };
   246→    const finish = (el: HTMLDivElement | null, ds: DragState, commit: boolean) => {
   247→      // Hareketsiz bırakma = tıklama; çevirme işini click olayına bırak
   248→      if (!dragMoved.current) {
   249→        dragRef.current = null;
   250→        setDrag(null);
   251→        return;
   252→      }
   253→      lastDragAt.current = Date.now();
   254→      const settled: DragState = { ...ds, settling: commit ? "commit" : "back" };
   255→      dragRef.current = settled;
   256→      setDrag(settled);
   257→      let fired = false;
   258→      const done = () => {
   259→        if (fired) return;
   260→        fired = true;
   261→        if (commit) setPage(ds.to);
   262→        setDrag(null);
   263→        dragRef.current = null;
   264→      };
   265→      if (!el) {
   266→        done();
   267→        return;
   268→      }
   269→      const current = leafTransform(ds);
   270→      let kf: Keyframe[];
   271→      let dur: number;
   272→      if (commit) {
   273→        dur = 400;
   274→        kf =
   275→          ds.dir === "next"
   276→            ? [
   277→                { transform: current, offset: 0 },
   278→                { transform: "rotateY(-169deg) rotateZ(1.5deg) scale(0.99, 1.012)", offset: 0.55 },
   279→                { transform: "rotateY(-181.5deg) rotateZ(0deg)", offset: 0.88 },
   280→                { transform: "rotateY(-180deg) rotateZ(0deg) translate3d(0,0,0) scale(1,1)", offset: 1 },
   281→              ]
   282→            : [
   283→                { transform: current, offset: 0 },
   284→                { transform: "rotateY(-8.5deg) rotateZ(-1.5deg) scale(0.99, 1.012)", offset: 0.55 },
   285→                { transform: "rotateY(1.5deg) rotateZ(0deg)", offset: 0.88 },
   286→                { transform: "rotateY(0deg) rotateZ(0deg) translate3d(0,0,0) scale(1,1)", offset: 1 },
   287→              ];
   288→      } else {
   289→        dur = 520;
   290→        kf =
   291→          ds.dir === "next"
   292→            ? [
   293→                { transform: current, offset: 0 },
   294→                { transform: "rotateY(7deg) rotateZ(-1.6deg)", offset: 0.55 },
   295→                { transform: "rotateY(-2.5deg) rotateZ(0.6deg)", offset: 0.82 },
   296→                { transform: "rotateY(0deg) rotateZ(0deg) translate3d(0,0,0) scale(1,1)", offset: 1 },
   297→              ]
   298→            : [
   299→                { transform: current, offset: 0 },
   300→                { transform: "rotateY(-187deg) rotateZ(1.6deg)", offset: 0.55 },
   301→                { transform: "rotateY(-177.5deg) rotateZ(-0.6deg)", offset: 0.82 },
   302→                { transform: "rotateY(-180deg) rotateZ(0deg) translate3d(0,0,0) scale(1,1)", offset: 1 },
   303→              ];
   304→      }
   305→      try {
   306→        const anim = el.animate(kf, {
   307→          duration: dur,
   308→          easing: "cubic-bezier(0.34, 0.9, 0.34, 1)",
   309→          fill: "forwards",
   310→        });
   311→        anim.finished.then(done).catch(done);
   312→      } catch {
   313→        done();
   314→      }
   315→      // Dış yarı (katlama) yaprakla birlikte eşzamanlı oturur — kâğıt bükülmesi
   316→      const foldEl = foldRef.current;
   317→      if (foldEl) {
   318→        const foldCurrent = foldTransform(ds);
   319→        const foldKf: Keyframe[] = commit
   320→          ? [
   321→              { transform: foldCurrent, offset: 0 },
   322→              { transform: "rotateY(-15deg)", offset: 0.5 },
   323→              { transform: "rotateY(0deg)", offset: 1 },
   324→            ]
   325→          : [
   326→              { transform: foldCurrent, offset: 0 },
   327→              { transform: "rotateY(-24deg)", offset: 0.5 },
   328→              { transform: "rotateY(0deg)", offset: 1 },
   329→            ];
   330→        try {
   331→          foldEl.animate(foldKf, {
   332→            duration: dur,
   333→            easing: "cubic-bezier(0.34, 0.9, 0.34, 1)",
   334→            fill: "forwards",
   335→          });
   336→        } catch {
   337→          /* katlama animasyonu opsiyonel */
   338→        }
   339→      }
   340→    };
   341→    const onUp = () => {
   342→      const ds = dragRef.current;
   343→      if (!ds || ds.settling) return;
   344→      finish(leafRef.current, ds, ds.p >= 0.34);
   345→    };
   346→    const onCancel = () => {
   347→      const ds = dragRef.current;
   348→      if (!ds || ds.settling) return;
   349→      finish(leafRef.current, ds, false);
   350→    };
   351→    window.addEventListener("pointermove", onMove, { passive: true });
   352→    window.addEventListener("pointerup", onUp);
   353→    window.addEventListener("pointercancel", onCancel);
   354→    return () => {
   355→      if (raf) cancelAnimationFrame(raf);
   356→      window.removeEventListener("pointermove", onMove);
   357→      window.removeEventListener("pointerup", onUp);
   358→      window.removeEventListener("pointercancel", onCancel);
   359→    };
   360→  }, [dragging]);
   361→
   362→  // ---- Sayfa etiketi ----
   363→  const pageLabel = useMemo(() => {
   364→    const p = pages[page];
   365→    if (!p) return "";
   366→    if (p.kind === "cover") return "Kapak";
   367→    if (p.kind === "toc") return "İçindekiler";
   368→    if (p.kind === "empty") return "Boş Sayı";
   369→    return `Sayfa ${p.folio}`;
   370→  }, [pages, page]);
   371→
   372→  // ---- Çevirme/sürükleme sırasında sahnede duracak sayfa ----
   373→  const staticIndex = drag
   374→    ? drag.dir === "next"
   375→      ? drag.to
   376→      : drag.from
   377→    : flip
   378→      ? flip.dir === "next"
   379→        ? flip.to
   380→        : flip.from
   381→      : page;
   382→  const leafIndex = drag
   383→    ? drag.dir === "next"
   384→      ? drag.from
   385→      : drag.to
   386→    : flip
   387→      ? flip.dir === "next"
   388→        ? flip.from
   389→        : flip.to
   390→      : -1;
   391→
   392→  // ================= SAYFA ŞABLONLARI =================
   393→
   394→  const renderRunningHead = (center: string, folio?: number) => (
   395→    <div className="flex items-center justify-between gap-3 border-b border-stone-900/20 pb-1.5 text-[9px] uppercase tracking-[0.24em] text-stone-600 sm:text-[10px]">
   396→      <span className="mag-serif font-bold tracking-[0.18em] text-stone-800">KRİMİSCAN GLOBAL</span>
   397→      <span className="min-w-0 flex-1 truncate text-center">{center}</span>
   398→      <span className="mag-serif">{folio ?? ""}</span>
   399→    </div>
   400→  );
   401→
   402→  const renderStars = (importance: number, size = "h-2.5 w-2.5") => (
   403→    <span className="inline-flex items-center gap-0.5" title={`Önem puanı: ${importance}/5`}>
   404→      {Array.from({ length: 5 }).map((_, i) => (
   405→        <Star
   406→          key={i}
   407→          className={`${size} ${i < importance ? "fill-amber-700 text-amber-700" : "text-stone-400"}`}
   408→        />
   409→      ))}
   410→    </span>
   411→  );
   412→
   413→  const renderCover = (p: Extract<MagPage, { kind: "cover" }>) => {
   414→    const featImg = coverImage(p.featured);
   415→    return (
   416→      <div className="mag-page-content flex h-full flex-col px-4 pb-4 pt-4 sm:px-7 sm:pb-6 sm:pt-6">
   417→        {/* Üst şerit */}
   418→        <div className="flex items-center justify-between border-b border-stone-900/25 pb-1.5 text-[8px] uppercase tracking-[0.2em] text-stone-700 sm:text-[9px]">
   419→          <span>Adli Bilimler & Olay Yeri İncelemesi</span>
   420→          <span className="hidden sm:inline">Türkiyenin Küresel İzleme Dergisi</span>
   421→          <span>{issue.year}</span>
   422→        </div>
   423→
   424→        {/* Ana manşet */}
   425→        <h1 className="mag-serif mt-2 text-[clamp(1.9rem,6vw,3.1rem)] font-black leading-[0.95] tracking-tight text-stone-900">
   426→          KRİMİSCAN <span className="text-amber-700">GLOBAL</span>
   427→        </h1>
   428→        <div className="mt-1.5 flex items-center justify-between border-y-2 border-stone-900/80 py-1 text-[8px] font-bold uppercase tracking-[0.18em] text-stone-800 sm:text-[10px]">
   429→          <span>
   430→            Sayı {issue.no} · {issue.month} {issue.year}
   431→          </span>
   432→          <span className="text-red-800">{p.totalCount} konu tarandı</span>
   433→          <span>Ücretsiz E-Dergi</span>
   434→        </div>
   435→
   436→        {/* Kapak görseli — tıklayınca tam makale okuma penceresi */}
   437→        <button
   438→          type="button"
   439→          onClick={() => setReading(p.featured)}
   440→          className="group relative mt-3 block w-full overflow-hidden rounded-sm border border-stone-900/30 focus:outline-none focus-visible:ring-2 focus-visible:ring-amber-700"
   441→          title="Kapak haberini Tam Oku ile oku"
   442→        >
   443→          {featImg ? (
   444→            /* eslint-disable-next-line @next/next/no-img-element */
   445→            <img
   446→              src={imgProxyUrl(featImg.url)}
   447→              alt={featImg.alt || p.featured.trTitle}
   448→              className="h-[clamp(140px,26vh,270px)] w-full object-cover transition-transform duration-500 group-hover:scale-[1.02]"
   449→              onError={(e) => (e.currentTarget.style.visibility = "hidden")}
   450→            />
   451→          ) : (
   452→            <div className="flex h-[clamp(120px,20vh,220px)] w-full items-center justify-center bg-stone-900/[0.06]">
   453→              <Fingerprint className="h-12 w-12 text-stone-400" />
   454→            </div>
   455→          )}
   456→          <span className="mag-tag-red absolute left-2 top-2">Kapak Haberi</span>
   457→        </button>
   458→
   459→        {/* Kapak başlığı */}
   460→        <h2 className="mag-serif mt-3 line-clamp-3 text-[clamp(1.05rem,2.9vw,1.65rem)] font-bold leading-tight text-stone-900">
   461→          {p.featured.trTitle}
   462→        </h2>
   463→        <p className="mt-1.5 line-clamp-2 text-[12px] leading-relaxed text-stone-700 sm:text-[13px]">
   464→          {clip(p.featured.trSummary, 210)}
   465→        </p>
   466→
   467→        {/* Yan başlıklar şeridi */}
   468→        <div className="mt-auto grid grid-cols-3 gap-3 border-t border-stone-900/20 pt-3">
   469→          {p.secondary.map((s) => {
   470→            const sImg = coverImage(s);
   471→            return (
   472→              <button
   473→                key={s.id}
   474→                type="button"
   475→                onClick={() => setReading(s)}
   476→                className="group text-left focus:outline-none"
   477→                title="Tam Oku ile aç"
   478→              >
   479→                <span className="mag-kicker block !text-[8px] sm:!text-[9px]">
   480→                  {categoryLabel(s.category).split(" ")[0]}
   481→                </span>
   482→                <span className="mag-serif mt-1 line-clamp-3 block text-[11px] font-semibold leading-snug text-stone-900 group-hover:text-amber-800 sm:text-[12.5px]">
   483→                  {s.trTitle}

   916→            <>
   917→              {page < totalPages - 1 && (
   918→                <button
   919→                  type="button"
   920→                  className="mag-ear mag-ear-right"
   921→                  onPointerDown={onEarPointerDown("next")}
   922→                  onClick={() => {
   923→                    if (Date.now() - lastDragAt.current < 600) return;
   924→                    goNext();
   925→                  }}
   926→                  aria-label="Sayfayı sağ alt köşeden tutup çevir"
   927→                  title="Sağ alt köşeden tutup çevir"
   928→                >
   929→                  <span className="mag-ear-fold" />
   930→                </button>
   931→              )}
   932→              {page > 0 && (
   933→                <button
   934→                  type="button"
   935→                  className="mag-ear mag-ear-left"
   936→                  onPointerDown={onEarPointerDown("prev")}
   937→                  onClick={() => {
   938→                    if (Date.now() - lastDragAt.current < 600) return;
   939→                    goPrev();
   940→                  }}
   941→                  aria-label="Sayfayı sol alt köşeden tutup çevir"
   942→                  title="Sol alt köşeden tutup çevir"
   943→                >
   944→                  <span className="mag-ear-fold" />
   945→                </button>

   947→            </>
   948→          )}
   949→        </div>
   950→
   951→        {/* Alt ilerleme çubuğu */}
   952→        <div className="mx-auto mt-4 flex w-full max-w-3xl items-center gap-3">
   953→          <div className="h-1 flex-1 overflow-hidden rounded-full bg-slate-800">
   954→            <div
   955→              className="h-full rounded-full bg-amber-500 transition-all duration-500"
   956→              style={{ width: `${((page + 1) / totalPages) * 100}%` }}
   957→            />
   958→          </div>
   959→          <span className="shrink-0 text-[11px] tabular-nums text-slate-400">
   960→            {page + 1} / {totalPages}
   961→          </span>
   962→        </div>
   963→      </div>
   964→
   965→      {/* Tam Oku — makalenin tamamı büyük okuma penceresinde */}
   966→      {reading && (
   967→        <FullReadDialog
   968→          finding={findings.find((x) => x.id === reading.id) ?? reading}
   969→          onClose={() => setReading(null)}
   970→          onToggleBookmark={onToggleBookmark}
   971→        />
   972→      )}
   973→    </section>
   974→  );

    18→export async function getZAI(): Promise<ZAI> {
    19→  if (zaiInstance) return zaiInstance;
    20→  if (!zaiInitPromise) {
    21→    zaiInitPromise = (async () => {
    22→      const baseUrl = process.env.ZAI_BASE_URL?.trim();
    23→      const apiKey = process.env.ZAI_API_KEY?.trim();
    24→      if (baseUrl && apiKey) {
    25→        // SDK kurucusu tip tanımında private olarak işaretlidir; ortam
    26→        // değişkenleriyle doğrudan yapılandırma için güvenli tip geçişi gerekir.
    27→        const Ctor = ZAI as unknown as new (config: Record<string, string>) => ZAI;
    28→        zaiInstance = new Ctor({
    29→          baseUrl,
    30→          apiKey,
    31→          chatId: process.env.ZAI_CHAT_ID?.trim() ?? "",
    32→          userId: process.env.ZAI_USER_ID?.trim() ?? "",
    33→          token: process.env.ZAI_TOKEN?.trim() ?? "",
    34→        });
    35→        return zaiInstance;
    36→      }
    37→      zaiInstance = await ZAI.create();
    38→      return zaiInstance;
    39→    })();
    40→    zaiInitPromise.catch(() => {
    41→      // Başarısız init sonrası tekrar denemeye izin ver
    42→      zaiInitPromise = null;
    43→    });
    44→  }
    45→  return zaiInitPromise;

    76→const DDG_UA =
    77→  "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/125.0.0.0 Safari/537.36";
    78→
    79→/**
    80→ * DuckDuckGo HTML araması (API anahtarı gerektirmez). Z-AI web_search
    81→ * kullanılamadığında yedek arama motoru olarak çalışır.
    82→ */
    83→export async function searchWebDuckDuckGo(query: string, num = 10): Promise<RawSearchItem[]> {
    84→  try {
    85→    const res = await fetch(
    86→      `https://html.duckduckgo.com/html/?q=${encodeURIComponent(query)}`,
    87→      {
    88→        headers: {
    89→          "User-Agent": DDG_UA,
    90→          Accept: "text/html,application/xhtml+xml",
    91→          "Accept-Language": "tr,en;q=0.8",
    92→        },
    93→        signal: AbortSignal.timeout(15000),
    94→        cache: "no-store",
    95→      }
    96→    );
    97→    if (!res.ok) return [];
    98→    const html = await res.text();
    99→
   100→    const anchorRe = /<a\s([^>]*class="[^"]*result__a[^"]*"[^>]*)>([\s\S]*?)<\/a>/g;
   101→    const snippetRe = /<a\s([^>]*class="[^"]*result__snippet[^"]*"[^>]*)>([\s\S]*?)<\/a>/g;
   102→
   103→    const snippets: string[] = [];
   104→    let sm: RegExpExecArray | null;
   105→    while ((sm = snippetRe.exec(html)) !== null) {
   106→      snippets.push(tagText(sm[2]).slice(0, 600));
   107→    }
   108→
   109→    const items: RawSearchItem[] = [];
   110→    let m: RegExpExecArray | null;
   111→    while ((m = anchorRe.exec(html)) !== null && items.length < num) {
   112→      const hrefMatch = m[1].match(/href="([^"]+)"/);
   113→      if (!hrefMatch) continue;
   114→      let href = hrefMatch[1].replace(/&amp;/g, "&");
   115→      if (href.startsWith("//")) href = `https:${href}`;
   116→      if (/duckduckgo\.com\/l\//i.test(href)) {
   117→        try {
   118→          const real = new URL(href).searchParams.get("uddg");
   119→          if (!real) continue;
   120→          href = real;
   121→        } catch {
   122→          continue;
   123→        }
   124→      }
   125→      if (!/^https?:\/\//i.test(href)) continue;
   126→      if (/duckduckgo\.com\/y\.js/i.test(href)) continue; // reklam bağlantısı
   127→      const title = tagText(m[2]);
   128→      if (!title || title.length < 3) continue;
   129→      let host = "";
   130→      try {
   131→        host = new URL(href).hostname;
   132→      } catch {
   133→        continue;
   134→      }
   135→      items.push({
   136→        url: href,
   137→        name: title,
   138→        snippet: snippets[items.length] ?? "",
   139→        host_name: host,
   140→        date: "",
   141→      });
   142→    }
   143→    return items;
   144→  } catch (err) {
   145→    console.error(`[searchWebDuckDuckGo] Arama hatası (${query}):`, err);
   146→    return [];
   147→  }

   180→/** LLM ile toplu çeviri + kategorizasyon. Hata durumunda kısmi sonuç döner. */
   181→export async function translateAndClassifyBatch(
   182→  items: TranslateInput[],
   183→  lang: ScanLanguage
   184→): Promise<TranslatedFinding[]> {
   185→  if (items.length === 0) return [];
   186→
   187→  const userPrompt = `Kaynak dil: ${lang.name} (${lang.code}).
   188→Aşağıdaki ${items.length} adet adli bilimler / olay yeri inceleme haberini Türkçe'ye çevir ve analiz et.
   189→Dikkat: Reklam, genel haber, alakasız içerik varsa category="genel", importance=1 ver.
   190→
   191→GIRDİ (JSON):
   192→${JSON.stringify(items, null, 1)}`;
   193→
   194→  for (let attempt = 1; attempt <= 2; attempt++) {
   195→    try {
   196→      const zai = await getZAI();
   197→      const completion = await zai.chat.completions.create({
   198→        messages: [
   199→          { role: "assistant", content: TRANSLATE_SYSTEM_PROMPT },
   200→          { role: "user", content: userPrompt },
   201→        ],
   202→        thinking: { type: "disabled" },
   203→      });
   204→
   205→      const raw = completion?.choices?.[0]?.message?.content ?? "";
   206→      const parsed = extractJsonArray(raw);
   207→      if (!parsed) throw new Error("JSON ayrıştırılamadı");
   208→
   209→      const valid = parsed
   210→        .filter((it: any) => typeof it?.index === "number" && it?.trTitle)
   211→        .map((it: any) => ({
   212→          index: it.index,
   213→          trTitle: String(it.trTitle).slice(0, 300),
   214→          trSummary: String(it.trSummary ?? "").slice(0, 1200),
   215→          category: CATEGORY_KEYS.includes(it.category) ? it.category : "genel",
   216→          importance: Math.min(5, Math.max(1, Number(it.importance) || 3)),
   217→          isNewTech: Boolean(it.isNewTech),
   218→          keywords: Array.isArray(it.keywords)
   219→            ? it.keywords.slice(0, 5).map((k: any) => String(k).slice(0, 40))
   220→            : [],
   221→        })) as TranslatedFinding[];
   222→
   223→      if (valid.length > 0) return valid;
   224→      throw new Error("Geçerli sonuç boş");
   225→    } catch (err) {
   226→      console.error(`[translateBatch] Deneme ${attempt} başarısız:`, err);
   227→      if (attempt === 2) {
   228→        // Fallback: çevrilmemiş ama kayıtlı olsun
   229→        return items.map((it) => ({
   230→          index: it.index,
   231→          trTitle: it.title.slice(0, 300),
   232→          trSummary: it.snippet.slice(0, 500) + " [Otomatik çeviri başarısız oldu]",
   233→          category: "genel",
   234→          importance: 2,
   235→          isNewTech: false,
   236→          keywords: [],
   237→        }));
   238→      }
   239→      await new Promise((r) => setTimeout(r, 1200));
   240→    }
   241→  }
   242→  return [];
   243→}

   245→/** Metin içinden JSON dizisini çıkarır (kod bloğu / ek met toleranslı) */
   246→function extractJsonArray(text: string): any[] | null {
   247→  if (!text) return null;
   248→  const cleaned = text.replace(/```json/gi, "```").trim();
   249→  const start = cleaned.indexOf("[");
   250→  const end = cleaned.lastIndexOf("]");
   251→  if (start === -1 || end === -1 || end <= start) return null;
   252→  try {
   253→    return JSON.parse(cleaned.slice(start, end + 1));
   254→  } catch {
   255→    return null;
   256→  }
   257→}

   371→/** HTML tag dizesinin iç metnini temizler */
   372→function tagText(html: string): string {
   373→  return html
   374→    .replace(/<script[\s\S]*?<\/script>/gi, " ")
   375→    .replace(/<style[\s\S]*?<\/style>/gi, " ")
   376→    .replace(/<[^>]+>/g, " ")
   377→    .replace(/&nbsp;/gi, " ")
   378→    .replace(/&amp;/gi, "&")
   379→    .replace(/&lt;/gi, "<")
   380→    .replace(/&gt;/gi, ">")
   381→    .replace(/&quot;/gi, '"')
   382→    .replace(/&#0?39;/gi, "'")
   383→    .replace(/&#x27;/gi, "'")
   384→    .replace(/\s+/g, " ")
   385→    .trim();
   386→}

   388→/**
   389→ * Makale HTML'inden paragraf yapısını koruyarak ana metni çıkarır:
   390→ * script/style/nav/header/footer/aside/form atılır, <p> blokları sırayla toplanır.
   391→ */
   392→export function extractArticleParagraphs(html: string, maxChars = FULL_TEXT_LIMIT): string[] {
   393→  if (!html) return [];
   394→  const cleaned = html
   395→    .replace(/<script[\s\S]*?<\/script>/gi, " ")
   396→    .replace(/<style[\s\S]*?<\/style>/gi, " ")
   397→    .replace(/<noscript[\s\S]*?<\/noscript>/gi, " ")
   398→    .replace(/<(nav|header|footer|aside|form|button|svg)[\s\S]*?<\/\1>/gi, " ");
   399→
   400→  const paras: string[] = [];
   401→  const pRe = /<p\s[^>]*>([\s\S]*?)<\/p\s*>/gi;
   402→  let m: RegExpExecArray | null;
   403→  let total = 0;
   404→  while ((m = pRe.exec(cleaned)) !== null) {
   405→    const text = tagText(m[1]);
   406→    // Menü/tanıtım cümlecikleri ve tek kelimelik başlıkları ele
   407→    if (text.length < 45 || text.split(/\s+/).length < 7) continue;
   408→    // Aşırı bağlantı dolu paragraflar genelde navigasyondur
   409→    const linkCount = (m[1].match(/<a\s/gi) ?? []).length;
   410→    if (linkCount > 4) continue;
   411→    paras.push(text);
   412→    total += text.length;
   413→    if (total >= maxChars) break;
   414→  }
   415→
   416→  // <p> yoksa (bazı sayfalarda metin div'lerde) tüm sayfa metnine düş
   417→  if (paras.length === 0) {
   418→    const all = stripHtml(cleaned);
   419→    if (all.length > 200) {
   420→      // Cümle sınırlarına göre kaba paragraf bölümlemesi
   421→      const sentences = all.split(/(?<=[.!?。!?])\s+/);
   422→      let buf = "";
   423→      for (const s of sentences) {
   424→        buf = buf ? `${buf} ${s}` : s;
   425→        if (buf.length >= 320) {
   426→          paras.push(buf.trim());
   427→          buf = "";
   428→          if (paras.join(" ").length >= maxChars) break;
   429→        }
   430→      }
   431→      if (buf.trim().length > 60) paras.push(buf.trim());
   432→    }
   433→  }
   434→
   435→  return paras.slice(0, 150);
   436→}

   465→function splitIntoChunks(paras: string[]): string[][] {
   466→  const chunks: string[][] = [];
   467→  let buf: string[] = [];
   468→  let size = 0;
   469→  for (const p of paras) {
   470→    if (size + p.length > CHUNK_LIMIT && buf.length > 0) {
   471→      chunks.push(buf);
   472→      buf = [];
   473→      size = 0;
   474→    }
   475→    buf.push(p);
   476→    size += p.length;
   477→  }
   478→  if (buf.length > 0) chunks.push(buf);
   479→  return chunks;
   480→}

   482→/** Bir parçayı (paragraf listesini) Türkçeye çevirir; paragraf sayısını korumaya çalışır */
   483→async function translateChunk(chunk: string[], langName: string): Promise<string[]> {
   484→  const prompt = `Aşağıda ${langName} dilinde bir haber makalesinden ${chunk.length} adet paragraf var.
   485→Her paragrafı profesyonel, akıcı TÜRKÇE'ye çevir (adli bilimler / olay yeri inceleme alanı).
   486→
   487→KURALLAR:
   488→- Girdideki paragraf SAYISINI ve SIRASINI aynen koru; paragrafları boş satırla ayır.
   489→- Başlık veya yorum EKLEME; yalnızca çeviriyi üret.
   490→- Teknik terimlerin Türkçe karşılığını ver, gerekirse parantezle orijinalini ekle.
   491→- Bir paragraf anlamsız/geçersizse onu kısa bir "(çevrilemedi)" ibaresiyle boş geçme, atla.
   492→
   493→PARAGRAFLAR:
   494→${chunk.map((p, i) => `--${i + 1}--\n${p}`).join("\n\n")}`;
   495→
   496→  for (let attempt = 1; attempt <= 2; attempt++) {
   497→    try {
   498→      const zai = await getZAI();
   499→      const completion = await zai.chat.completions.create({
   500→        messages: [
   501→          {
   502→            role: "assistant",
   503→            content:
   504→              "Sen adli bilimler alanında uzmanlaşmış profesyonel bir çevirmensin. Yalnızca istenen çeviriyi, paragraf yapısını koruyarak üretirsin.",
   505→          },
   506→          { role: "user", content: prompt },
   507→        ],
   508→        thinking: { type: "disabled" },
   509→      });
   510→      const raw = completion?.choices?.[0]?.message?.content ?? "";
   511→      const parts = raw
   512→        .replace(/```[a-z]*\n?/gi, "")
   513→        .split(/\n{2,}/)
   514→        .map((s) => s.replace(/^--\d+--\s*/g, "").trim())
   515→        .filter((s) => s.length > 2);
   516→      if (parts.length >= Math.max(1, Math.floor(chunk.length * 0.6))) return parts;
   517→      throw new Error(`Paragraf sayısı uyuşmadı (${parts.length}/${chunk.length})`);
   518→    } catch (err) {
   519→      console.error(`[translateChunk] Deneme ${attempt} başarısız:`, err);
   520→      if (attempt === 2) {
   521→        // Çevrilemeyen parçanın orijinali kalsın (kullanıcı yine de okuyabilsin)
   522→        return chunk.map((p) => `${p} *(çevrilemedi)*`);
   523→      }
   524→      await new Promise((r) => setTimeout(r, 1000));
   525→    }
   526→  }
   527→  return chunk;
   528→}

   530→/**
   531→ * Tam Oku: makalenin TAM metnini kaynaktan okur ve Türkçeye çevirir.
   532→ * Kaynak dil Türkçeyse çeviri atlanır. Hata durumunda mümkün olan kısmı döndürür.
   533→ */
   534→export async function fetchAndTranslateFullArticle(
   535→  url: string,
   536→  originalTitle: string,
   537→  trTitle: string,
   538→  language: string
   539→): Promise<FullArticleResult> {
   540→  const { html, engine } = await fetchPageHtmlForText(url);
   541→
   542→  if (engine === "none" || !html) {
   543→    // Sayfa okunamadı — boş sonuç; çağıran taraf özete düşer
   544→    return { title: trTitle, paragraphs: [], wordCount: 0, engine: "none", translated: false };
   545→  }
   546→
   547→  const sourceParas = extractArticleParagraphs(html);
   548→  const isTurkish = /^tr\b/i.test(language) || /türk/i.test(language);
   549→
   550→  if (isTurkish) {
   551→    return {
   552→      title: trTitle,
   553→      paragraphs: sourceParas,
   554→      wordCount: sourceParas.join(" ").split(/\s+/).filter(Boolean).length,
   555→      engine,
   556→      translated: false,
   557→    };
   558→  }
   559→
   560→  if (sourceParas.length === 0) {
   561→    return { title: trTitle, paragraphs: [], wordCount: 0, engine, translated: false };
   562→  }
   563→
   564→  const chunks = splitIntoChunks(sourceParas);
   565→  const translatedParts: string[] = [];
   566→  for (const chunk of chunks) {
   567→    const done = await translateChunk(chunk, language);
   568→    translatedParts.push(...done);
   569→  }
   570→
   571→  const paragraphs = translatedParts.filter((p) => p && p.trim()).slice(0, 150);
   572→  return {
   573→    title: trTitle,
   574→    paragraphs,
   575→    wordCount: paragraphs.join(" ").split(/\s+/).filter(Boolean).length,
   576→    engine,
   577→    translated: true,
   578→  };
   579→}

   671→/**
   672→ * AI İstihbarat Asistanı: arşiv bağlamı + güncel web araması ile Türkçe yanıt üretir.
   673→ */
   674→export async function assistantReply(
   675→  history: AssistantMessage[],
   676→  archiveContext: string,
   677→  webContext: string
   678→): Promise<string> {
   679→  const lastUser =
   680→    [...history].reverse().find((m) => m.role === "user")?.content ?? "";
   681→
   682→  const systemPrompt = `Sen OYİ GLOBAL'in AI İstihbarat Asistanısın: olay yeri inceleme teknikleri ve adli bilimler alanında uzmanlaşmış, TÜRKÇE konuşan bir analistsin.
   683→
   684→GÖREVİN:
   685→- Kullanıcının sorularını, verilen ARŞİV BAĞLAMI ve gerekirse GENEL BİLGİNle yanıtla.
   686→- Arşivde ilgili bulgu varsa bunlara [B#n] koduyla atıf yap ve kaynak adını söyle.
   687→- Küresel perspektif ver: hangi ülke/kurum ne yapıyor, hangi yöntem öne çıkıyor.
   688→- Teknik terimlerin Türkçe karşılığını parantez içinde orijinaliyle ver (örn. "iz delili (trace evidence)").
   689→- Yanıtlarını markdown ile yapılandır: kısa paragraflar, gerektiğinde madde işaretleri.
   690→- Alanla ilgisiz sorularda kibarca alanına yönlendir.
   691→- Emin olmadığın bilgiyi uydurma; "kesin bilgi bulunamadı" deyip kaynak öner.
   692→
   693→${archiveContext}
   694→${webContext}`;
   695→
   696→  const messages: { role: "assistant" | "user"; content: string }[] = [
   697→    { role: "assistant", content: systemPrompt },
   698→    // Son 8 mesajı koru (bağlam penceresi)
   699→    ...history.slice(-8).map((m) => ({ role: m.role, content: m.content })),
   700→  ];
   701→
   702→  for (let attempt = 1; attempt <= 2; attempt++) {
   703→    try {
   704→      const zai = await getZAI();
   705→      const completion = await zai.chat.completions.create({
   706→        messages,
   707→        thinking: { type: "disabled" },
   708→      });
   709→      const content = completion?.choices?.[0]?.message?.content ?? "";
   710→      if (content.trim().length > 2) return content.trim();
   711→      throw new Error("Boş yanıt");
   712→    } catch (err) {
   713→      console.error(`[assistantReply] Deneme ${attempt} başarısız:`, err);
   714→      if (attempt === 2) {
   715→        return `## ⚠️\nYanıt üretilirken bir sorun oluştu. Lütfen sorunuzu tekrar iletin.\n\n*(Soru: "${lastUser.slice(0, 120)}")*`;
   716→      }
   717→      await new Promise((r) => setTimeout(r, 1200));
   718→    }
   719→  }
   720→  return "Yanıt üretilemedi, lütfen tekrar deneyin.";
   721→}

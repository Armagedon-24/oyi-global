     1→"use client";
     2→
     3→/**
     4→ * OYİ GLOBAL — Yönetim Paneli (/admin)
     5→ * Şifreli giriş (ADMIN_PASSWORD, .env) → dergi taraması, bulgu yönetimi,
     6→ * yorum moderasyonu. Okuma yüzeyi / (dergi) sayfasında kalır; tarama
     7→ * yalnızca bu panel üzerinden yapılabilir.
     8→ */
     9→
    10→import { useCallback, useEffect, useState } from "react";
    11→import Link from "next/link";
    12→import { Button } from "@/components/ui/button";
    13→import { Card, CardContent } from "@/components/ui/card";
    14→import { Input } from "@/components/ui/input";
    15→import { Badge } from "@/components/ui/badge";
    16→import { Progress } from "@/components/ui/progress";
    17→import { ScrollArea } from "@/components/ui/scroll-area";
    18→import {
    19→  Select,
    20→  SelectContent,
    21→  SelectItem,
    22→  SelectTrigger,
    23→  SelectValue,
    24→} from "@/components/ui/select";
    25→import { useToast } from "@/hooks/use-toast";
    26→import {
    27→  CircleAlert,
    28→  CircleCheck,
    29→  Database,
    30→  ExternalLink,
    31→  Globe2,
    32→  Languages,
    33→  Loader2,
    34→  Lock,
    35→  LogOut,
    36→  MessageSquare,
    37→  Radar,
    38→  Radar as RadarIcon,
    39→  ScanSearch,
    40→  ShieldCheck,
    41→  Sparkles,
    42→  Trash2,
    43→} from "lucide-react";
    44→import { CATEGORIES, SCAN_LANGUAGES } from "@/lib/forensics-config";
    45→import { OyiMark } from "@/components/forensics/oyi-logo";
    46→import type { FindingItem, ScanLogEntry, StatsData } from "@/lib/types";
    47→
    48→interface AdminComment {
    49→  id: string;
    50→  author: string;
    51→  body: string;
    52→  createdAt: string;
    53→  findingId: string;
    54→  findingTitle: string;
    55→}
    56→
    57→type AuthState = "checking" | "guest" | "admin";
    58→
    59→export default function AdminPage() {
    60→  const { toast } = useToast();
    61→
    62→  const [auth, setAuth] = useState<AuthState>("checking");
    63→  const [password, setPassword] = useState("");
    64→  const [loggingIn, setLoggingIn] = useState(false);
    65→  const [loginError, setLoginError] = useState<string | null>(null);
    66→
    67→  const [stats, setStats] = useState<StatsData | null>(null);
    68→  const [findings, setFindings] = useState<FindingItem[]>([]);
    69→  const [comments, setComments] = useState<AdminComment[]>([]);
    70→  const [commentTotal, setCommentTotal] = useState(0);
    71→  const [deletingFindingId, setDeletingFindingId] = useState<string | null>(null);
    72→  const [deletingCommentId, setDeletingCommentId] = useState<string | null>(null);
    73→
    74→  // ---- Tarama durumu ----
    75→  const [scanning, setScanning] = useState(false);
    76→  const [scope, setScope] = useState<string>("all");
    77→  const [focusCategory, setFocusCategory] = useState<string>("all");
    78→  const [totalLangs, setTotalLangs] = useState(0);
    79→  const [doneLangs, setDoneLangs] = useState(0);
    80→  const [currentLang, setCurrentLang] = useState<{ flag: string; name: string } | null>(null);
    81→  const [log, setLog] = useState<ScanLogEntry[]>([]);
    82→  const [sessionAdded, setSessionAdded] = useState(0);
    83→
    84→  const loadStats = useCallback(async () => {
    85→    try {
    86→      const res = await fetch("/api/stats");
    87→      const data = await res.json();
    88→      if (data.success) setStats(data.stats);
    89→    } catch {
    90→      /* sessiz */
    91→    }
    92→  }, []);
    93→
    94→  const loadFindings = useCallback(async () => {
    95→    try {
    96→      const res = await fetch("/api/findings?limit=15&sort=recent");
    97→      const data = await res.json();
    98→      if (data.success) setFindings(data.findings);
    99→    } catch {
   100→      /* sessiz */
   101→    }
   102→  }, []);
   103→
   104→  const loadComments = useCallback(async () => {
   105→    try {
   106→      const res = await fetch("/api/comments?limit=40");
   107→      if (res.status === 401) return;
   108→      const data = await res.json();
   109→      if (data.success) {
   110→        setComments(data.comments);
   111→        setCommentTotal(data.total ?? data.comments.length);
   112→      }
   113→    } catch {
   114→      /* sessiz */
   115→    }
   116→  }, []);
   117→
   118→  useEffect(() => {
   119→    fetch("/api/admin/session")
   120→      .then((r) => r.json())
   121→      .then((data) => setAuth(data.authenticated ? "admin" : "guest"))
   122→      .catch(() => setAuth("guest"));
   123→  }, []);
   124→
   125→  useEffect(() => {
   126→    if (auth !== "admin") return;
   127→    loadStats();
   128→    loadFindings();
   129→    loadComments();
   130→  }, [auth, loadStats, loadFindings, loadComments]);
   131→
   132→  const login = async (e: React.FormEvent) => {
   133→    e.preventDefault();
   134→    setLoggingIn(true);
   135→    setLoginError(null);
   136→    try {
   137→      const res = await fetch("/api/admin/login", {
   138→        method: "POST",
   139→        headers: { "Content-Type": "application/json" },
   140→        body: JSON.stringify({ password }),
   141→      });
   142→      const data = await res.json();
   143→      if (data.success) {
   144→        setPassword("");
   145→        setAuth("admin");
   146→      } else {
   147→        setLoginError(data.error || "Giriş başarısız.");
   148→      }
   149→    } catch {
   150→      setLoginError("Bağlantı hatası — giriş yapılamadı.");
   151→    } finally {
   152→      setLoggingIn(false);
   153→    }
   154→  };
   155→
   156→  const logout = async () => {
   157→    await fetch("/api/admin/logout", { method: "POST" }).catch(() => {});
   158→    setAuth("guest");
   159→  };
   160→
   161→  // ---- Tarama akışı (yalnızca bu panelde) ----
   162→  const startScan = async () => {
   163→    if (scanning) return;
   164→    setScanning(true);
   165→    setLog([]);
   166→    setSessionAdded(0);
   167→    setDoneLangs(0);
   168→    setCurrentLang(null);
   169→
   170→    try {
   171→      const startRes = await fetch("/api/scan/start", {
   172→        method: "POST",
   173→        headers: { "Content-Type": "application/json" },
   174→        body: JSON.stringify({ scope, focusCategory }),
   175→      });
   176→      const startData = await startRes.json();
   177→      if (!startData.success) throw new Error(startData.error || "Tarama başlatılamadı");
   178→
   179→      const scanId: string = startData.scanId;
   180→      const langs: { code: string; name: string; flag: string }[] = startData.languages;
   181→      setTotalLangs(langs.length);
   182→      let addedTotal = 0;
   183→
   184→      for (let i = 0; i < langs.length; i++) {
   185→        const lang = langs[i];
   186→        setCurrentLang({ flag: lang.flag, name: lang.name });
   187→        const logId = `${lang.code}-${Date.now()}`;
   188→        setLog((prev) => [
   189→          ...prev,
   190→          { id: logId, flag: lang.flag, langName: lang.name, status: "searching" },
   191→        ]);
   192→
   193→        try {
   194→          const res = await fetch("/api/scan/step", {
   195→            method: "POST",
   196→            headers: { "Content-Type": "application/json" },
   197→            body: JSON.stringify({ scanId, langCode: lang.code, focusCategory }),
   198→          });
   199→          const data = await res.json();
   200→          if (data.success) {
   201→            addedTotal += data.added;
   202→            setSessionAdded(addedTotal);
   203→            setLog((prev) =>
   204→              prev.map((e) =>
   205→                e.id === logId
   206→                  ? { ...e, status: "done", searched: data.searched, relevant: data.relevant, added: data.added }
   207→                  : e
   208→              )
   209→            );
   210→          } else {
   211→            setLog((prev) => prev.map((e) => (e.id === logId ? { ...e, status: "error" } : e)));
   212→          }
   213→        } catch {
   214→          setLog((prev) => prev.map((e) => (e.id === logId ? { ...e, status: "error" } : e)));
   215→        }
   216→        setDoneLangs(i + 1);
   217→      }
   218→
   219→      await fetch("/api/scan/complete", {
   220→        method: "POST",
   221→        headers: { "Content-Type": "application/json" },
   222→        body: JSON.stringify({ scanId }),
   223→      });
   224→
   225→      loadFindings();
   226→      loadStats();
   227→      loadComments();
   228→      toast({
   229→        title: "Küresel tarama tamamlandı",
   230→        description: `${langs.length} dil tarandı, ${addedTotal} yeni bulgu Türkçeleştirilerek dergiye eklendi.`,
   231→      });
   232→    } catch {
   233→      toast({
   234→        title: "Tarama başlatılamadı",
   235→        description: "Oturum süresi dolmuş olabilir; tekrar giriş yapmayı deneyin.",
   236→        variant: "destructive",
   237→      });
   238→    } finally {
   239→      setScanning(false);
   240→      setCurrentLang(null);
   241→    }
   242→  };
   243→
   244→  const deleteFinding = async (f: FindingItem) => {
   245→    setDeletingFindingId(f.id);
   246→    try {
   247→      const res = await fetch(`/api/findings/${f.id}`, { method: "DELETE" });
   248→      const data = await res.json();
   249→      if (data.success) {
   250→        setFindings((prev) => prev.filter((x) => x.id !== f.id));
   251→        loadStats();
   252→        toast({ title: "Bulgu silindi", description: f.trTitle.slice(0, 80) });
   253→      } else {
   254→        toast({ title: "Silinemedi", description: data.error, variant: "destructive" });
   255→      }
   256→    } catch {
   257→      toast({ title: "Silinemedi", description: "Bağlantı hatası.", variant: "destructive" });
   258→    } finally {
   259→      setDeletingFindingId(null);
   260→    }
   261→  };
   262→
   263→  const deleteComment = async (c: AdminComment) => {
   264→    setDeletingCommentId(c.id);
   265→    try {
   266→      const res = await fetch(`/api/comments/${c.id}`, { method: "DELETE" });
   267→      const data = await res.json();
   268→      if (data.success) {
   269→        setComments((prev) => prev.filter((x) => x.id !== c.id));
   270→        setCommentTotal((t) => Math.max(0, t - 1));
   271→      } else {
   272→        toast({ title: "Silinemedi", description: data.error, variant: "destructive" });
   273→      }
   274→    } catch {
   275→      toast({ title: "Silinemedi", description: "Bağlantı hatası.", variant: "destructive" });
   276→    } finally {
   277→      setDeletingCommentId(null);
   278→    }
   279→  };
   280→
   281→  // ================= GİRİŞ EKRANI =================
   282→  if (auth === "checking") {
   283→    return (
   284→      <div className="flex min-h-screen items-center justify-center bg-slate-950 text-slate-100">
   285→        <Loader2 className="h-6 w-6 animate-spin text-amber-400" />
   286→      </div>
   287→    );
   288→  }
   289→
   290→  if (auth === "guest") {
   291→    return (
   292→      <div className="flex min-h-screen flex-col items-center justify-center gap-6 bg-slate-950 px-4 text-slate-100">
   293→        <div className="flex items-center gap-3">
   294→          <OyiMark className="h-14 w-14" />
   295→          <div>
   296→            <h1 className="mag-serif text-xl font-black tracking-tight">
   297→              OYİ <span className="text-amber-400">GLOBAL</span>
   298→            </h1>
   299→            <p className="text-xs text-slate-400">Yönetim Paneli — Editör Girişi</p>
   300→          </div>
   301→        </div>
   302→
   303→        <form
   304→          onSubmit={login}
   305→          className="w-full max-w-sm space-y-3 rounded-xl border border-slate-800 bg-slate-900/70 p-6 shadow-xl"
   306→        >
   307→          <div className="flex items-center gap-2 text-sm font-semibold text-slate-200">
   308→            <Lock className="h-4 w-4 text-amber-400" />
   309→            Yönetici şifresi
   310→          </div>
   311→          <Input
   312→            type="password"
   313→            value={password}
   314→            onChange={(e) => setPassword(e.target.value)}
   315→            placeholder="Şifre"
   316→            autoFocus
   317→            className="bg-slate-950 border-slate-700"
   318→          />
   319→          {loginError && <p className="text-xs font-semibold text-red-400">{loginError}</p>}
   320→          <Button
   321→            type="submit"
   322→            disabled={loggingIn || password.length === 0}
   323→            className="w-full bg-amber-500 hover:bg-amber-400 text-slate-950 font-bold"
   324→          >
   325→            {loggingIn ? <Loader2 className="h-4 w-4 animate-spin" /> : <ShieldCheck className="h-4 w-4" />}
   326→            Panele Giriş Yap
   327→          </Button>
   328→          <p className="text-[11px] leading-snug text-slate-500">
   329→            Şifre projenin <code className="text-slate-400">.env</code> dosyasındaki{" "}
   330→            <code className="text-amber-400/90">ADMIN_PASSWORD</code> ile belirlenir.
   331→          </p>
   332→        </form>
   333→
   334→        <Link href="/" className="text-xs text-slate-400 underline-offset-2 hover:text-amber-300 hover:underline">
   335→          ← Dergiye geri dön
   336→        </Link>
   337→      </div>
   338→    );
   339→  }
   340→
   341→  // ================= PANEL =================
   342→  const lastScan = stats?.lastScan;
   343→  const lastScanDate = lastScan
   344→    ? new Date(lastScan.startedAt).toLocaleString("tr-TR", {
   345→        day: "numeric",
   346→        month: "short",
   347→        hour: "2-digit",
   348→        minute: "2-digit",
   349→      })
   350→    : null;
   351→
   352→  return (
   353→    <div className="min-h-screen bg-slate-950 text-slate-100">
   354→      {/* Üst bar */}
   355→      <header className="sticky top-0 z-30 border-b border-slate-800/80 bg-slate-950/90 backdrop-blur-md">
   356→        <div className="mx-auto flex h-16 max-w-6xl items-center justify-between gap-3 px-4 sm:px-6">
   357→          <div className="flex items-center gap-3 min-w-0">
   358→            <OyiMark className="h-10 w-10 shrink-0" />
   359→            <div className="min-w-0">
   360→              <h1 className="mag-serif font-black text-base leading-tight tracking-tight text-slate-50">
   361→                OYİ <span className="text-amber-400">GLOBAL</span>
   362→                <span className="ml-2 rounded-sm border border-emerald-500/40 bg-emerald-500/10 px-1.5 py-0.5 align-middle text-[9px] font-bold uppercase tracking-[0.18em] text-emerald-300">
   363→                  Yönetim
   364→                </span>
   365→              </h1>
   366→              <p className="text-[11px] text-slate-400 truncate">
   367→                Dergi taraması, içerik ve yorum yönetimi
   368→              </p>
   369→            </div>
   370→          </div>
   371→          <div className="flex items-center gap-2">
   372→            <Link
   373→              href="/"
   374→              className="inline-flex h-9 items-center gap-1.5 rounded-md border border-slate-700 bg-slate-900 px-3 text-sm text-slate-300 transition-colors hover:border-amber-500/50 hover:text-amber-300"
   375→            >
   376→              <ExternalLink className="h-3.5 w-3.5" />
   377→              <span className="hidden sm:inline">Dergiye Git</span>
   378→            </Link>
   379→            <Button
   380→              variant="outline"
   381→              onClick={logout}
   382→              className="h-9 border-slate-700 bg-slate-900 px-3 text-slate-300 hover:bg-slate-800 hover:text-red-300"
   383→            >
   384→              <LogOut className="h-3.5 w-3.5" />
   385→              <span className="hidden sm:inline">Çıkış</span>
   386→            </Button>
   387→          </div>
   388→        </div>
   389→      </header>
   390→
   391→      <main className="mx-auto w-full max-w-6xl space-y-6 px-4 py-6 sm:px-6">
   392→        {/* İstatistikler */}
   393→        <section className="grid grid-cols-2 lg:grid-cols-5 gap-3">
   394→          {[
   395→            { icon: Database, label: "Toplam Bulgu", value: stats?.total ?? 0, color: "text-amber-400" },
   396→            { icon: Sparkles, label: "Yeni Teknoloji", value: stats?.newTech ?? 0, color: "text-emerald-400" },
   397→            { icon: Languages, label: "Kaynak Dil", value: stats?.languages.length ?? 0, color: "text-purple-400" },
   398→            { icon: MessageSquare, label: "Okur Yorumu", value: commentTotal, color: "text-sky-400" },
   399→            { icon: ShieldCheck, label: "Son Güncelleme", value: lastScanDate ?? "—", color: "text-slate-200", small: true },
   400→          ].map((s, i) => (
   401→            <Card key={i} className="border-slate-800 bg-slate-900/70">
   402→              <CardContent className="p-4 flex items-center gap-3">
   403→                <div className="h-9 w-9 rounded-lg bg-slate-800/80 border border-slate-700 flex items-center justify-center shrink-0">
   404→                  <s.icon className={`h-4 w-4 ${s.color}`} />
   405→                </div>
   406→                <div className="min-w-0">
   407→                  <p className={`font-bold leading-none ${s.small ? "text-[13px] truncate" : "text-xl"}`}>
   408→                    {s.value}
   409→                  </p>
   410→                  <p className="text-[11px] text-slate-400 mt-1">{s.label}</p>
   411→                </div>
   412→              </CardContent>
   413→            </Card>
   414→          ))}
   415→        </section>
   416→
   417→        {/* Tarama paneli */}
   418→        <section className="rounded-2xl border border-slate-800 bg-gradient-to-b from-slate-900 to-slate-950 p-5 sm:p-6 space-y-4">
   419→          <div className="flex flex-wrap items-center justify-between gap-2">
   420→            <h2 className="flex items-center gap-2 font-semibold text-slate-100">
   421→              <Radar className="h-4 w-4 text-amber-400" />
   422→              Küresel Tarama — Yalnızca Editör
   423→            </h2>
   424→            <Badge className="bg-emerald-500/15 text-emerald-300 border-emerald-500/30 gap-1 hover:bg-emerald-500/15">
   425→              <ShieldCheck className="h-3 w-3" /> Admin yetkisi aktif
   426→            </Badge>
   427→          </div>
   428→          <p className="max-w-3xl text-xs leading-relaxed text-slate-400">
   429→            Tarama; {SCAN_LANGUAGES.length} dilde kaynak arar, adli bilimler ve olay yeri inceleme
   430→            gelişmelerini yakalar, yapay zekâ ile Türkçeye çevirip dergiye ekler. Bu işlem artık
   431→            herkese açık sayfalarda değil, yalnızca bu panelde çalıştırılabilir.
   432→          </p>
   433→
   434→          <div className="grid gap-3 sm:grid-cols-2 lg:grid-cols-4">
   435→            <div className="space-y-1.5">
   436→              <label className="text-xs font-medium text-slate-400">Tarama Kapsamı</label>
   437→              <Select value={scope} onValueChange={setScope} disabled={scanning}>
   438→                <SelectTrigger className="w-full bg-slate-900 border-slate-700 text-sm">
   439→                  <SelectValue />
   440→                </SelectTrigger>
   441→                <SelectContent className="bg-slate-900 border-slate-700">
   442→                  <SelectItem value="all">Tüm Diller ({SCAN_LANGUAGES.length} dil)</SelectItem>
   443→                  <SelectItem value="quick">Hızlı Tarama (6 dil)</SelectItem>
   444→                </SelectContent>
   445→              </Select>
   446→            </div>
   447→            <div className="space-y-1.5">
   448→              <label className="text-xs font-medium text-slate-400">Kategori Odağı</label>
   449→              <Select value={focusCategory} onValueChange={setFocusCategory} disabled={scanning}>
   450→                <SelectTrigger className="w-full bg-slate-900 border-slate-700 text-sm">
   451→                  <SelectValue />
   452→                </SelectTrigger>
   453→                <SelectContent className="bg-slate-900 border-slate-700">
   454→                  <SelectItem value="all">Tüm Kategoriler</SelectItem>
   455→                  {CATEGORIES.map((c) => (
   456→                    <SelectItem key={c.key} value={c.key}>
   457→                      {c.label}
   458→                    </SelectItem>
   459→                  ))}
   460→                </SelectContent>
   461→              </Select>
   462→            </div>
   463→            <div className="sm:col-span-2 flex items-end">
   464→              <Button
   465→                onClick={startScan}
   466→                disabled={scanning}
   467→                className="w-full h-10 bg-amber-500 hover:bg-amber-400 text-slate-950 font-bold text-sm"
   468→              >
   469→                {scanning ? (
   470→                  <>
   471→                    <Loader2 className="h-4 w-4 mr-2 animate-spin" /> Tarama sürüyor…
   472→                  </>
   473→                ) : (
   474→                  <>
   475→                    <ScanSearch className="h-4 w-4 mr-2" /> Küresel Taramayı Başlat
   476→                  </>
   477→                )}
   478→              </Button>
   479→            </div>
   480→          </div>
   481→
   482→          {/* Canlı ilerleme */}
   483→          {(scanning || log.length > 0) && (
   484→            <div className="grid gap-4 lg:grid-cols-3 pt-1">
   485→              <Card className="border-slate-800 bg-slate-900/70 lg:col-span-1">
   486→                <CardContent className="p-5 space-y-4">
   487→                  <div className="flex items-center justify-between">
   488→                    <h3 className="font-semibold text-sm text-slate-200 flex items-center gap-2">
   489→                      <RadarIcon className={`h-4 w-4 text-amber-400 ${scanning ? "animate-pulse" : ""}`} />
   490→                      Tarama Durumu
   491→                    </h3>
   492→                    <Badge
   493→                      className={
   494→                        scanning
   495→                          ? "bg-amber-500/15 text-amber-300 border-amber-500/30"
   496→                          : "bg-emerald-500/15 text-emerald-300 border-emerald-500/30"
   497→                      }
   498→                    >
   499→                      {scanning ? "Aktif" : "Tamamlandı"}
   500→                    </Badge>
   501→                  </div>
   502→
   503→                  {currentLang && (
   504→                    <div className="flex items-center gap-3 p-3 rounded-lg border border-amber-500/25 bg-amber-500/5">

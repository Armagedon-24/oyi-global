     9→
    10→import { useCallback, useEffect, useRef, useState } from "react";
    11→import Link from "next/link";
    12→import { Button } from "@/components/ui/button";
    13→import { Card, CardContent } from "@/components/ui/card";
    14→import { Input } from "@/components/ui/input";
    15→import { Textarea } from "@/components/ui/textarea";
    16→import { Badge } from "@/components/ui/badge";
    17→import { Progress } from "@/components/ui/progress";
    18→import {
    19→  Select,
    20→  SelectContent,
    21→  SelectItem,
    22→  SelectTrigger,
    23→  SelectValue,
    24→} from "@/components/ui/select";
    25→import { useToast } from "@/hooks/use-toast";
    26→import {
    27→  CircleAlert,
    28→  CircleCheck,
    29→  Database,
    30→  ExternalLink,
    31→  Globe2,
    32→  ImagePlus,
    33→  Languages,
    34→  Loader2,
    35→  Lock,
    36→  LogOut,
    37→  MessageSquare,
    38→  Newspaper,
    39→  Radar,
    40→  Radar as RadarIcon,
    41→  ScanSearch,
    42→  Send,
    43→  ShieldCheck,
    44→  Sparkles,
    45→  Trash2,
    46→  X,
    47→} from "lucide-react";
    48→import { CATEGORIES, SCAN_LANGUAGES } from "@/lib/forensics-config";
    49→import { OyiMark } from "@/components/forensics/oyi-logo";
    50→import type { FindingItem, ScanLogEntry, StatsData } from "@/lib/types";
    51→
    52→interface AdminComment {
    53→  id: string;
    54→  author: string;
    55→  body: string;
    56→  createdAt: string;
    57→  findingId: string;
    58→  findingTitle: string;
    59→}
    60→
    61→type AuthState = "checking" | "guest" | "admin";
    62→
    63→interface NewsImage {
    64→  url: string;
    65→  alt: string;
    66→  preview: string;
    67→}
    68→
    69→export default function AdminPage() {
    70→  const { toast } = useToast();
    71→
    72→  const [auth, setAuth] = useState<AuthState>("checking");
    73→  const [password, setPassword] = useState("");
    74→  const [loggingIn, setLoggingIn] = useState(false);
    75→  const [loginError, setLoginError] = useState<string | null>(null);
    76→
    77→  const [stats, setStats] = useState<StatsData | null>(null);
    78→  const [findings, setFindings] = useState<FindingItem[]>([]);
    79→  const [comments, setComments] = useState<AdminComment[]>([]);
    80→  const [commentTotal, setCommentTotal] = useState(0);
    81→  const [deletingFindingId, setDeletingFindingId] = useState<string | null>(null);
    82→  const [deletingCommentId, setDeletingCommentId] = useState<string | null>(null);
    83→
    84→  // ---- Yeni haber formu ----
    85→  const [newsTitle, setNewsTitle] = useState("");
    86→  const [newsBody, setNewsBody] = useState("");
    87→  const [newsCategory, setNewsCategory] = useState("genel");
    88→  const [newsSource, setNewsSource] = useState("");
    89→  const [newsImportance, setNewsImportance] = useState("4");
    90→  const [newsNewTech, setNewsNewTech] = useState(false);
    91→  const [newsImages, setNewsImages] = useState<NewsImage[]>([]);
    92→  const [uploading, setUploading] = useState(false);
    93→  const [savingNews, setSavingNews] = useState(false);
    94→  const fileInputRef = useRef<HTMLInputElement | null>(null);
    95→
    96→  // ---- Tarama durumu ----
    97→  const [scanning, setScanning] = useState(false);
    98→  const [scope, setScope] = useState<string>("all");
    99→  const [focusCategory, setFocusCategory] = useState<string>("all");
   100→  const [totalLangs, setTotalLangs] = useState(0);
   101→  const [doneLangs, setDoneLangs] = useState(0);
   102→  const [currentLang, setCurrentLang] = useState<{ flag: string; name: string } | null>(null);
   103→  const [log, setLog] = useState<ScanLogEntry[]>([]);
   104→  const [sessionAdded, setSessionAdded] = useState(0);
   105→
   106→  const loadStats = useCallback(async () => {
   107→    try {
   108→      const res = await fetch("/api/stats");
   109→      const data = await res.json();
   110→      if (data.success) setStats(data.stats);
   111→    } catch {
   112→      /* sessiz */
   113→    }
   114→  }, []);
   115→
   116→  const loadFindings = useCallback(async () => {
   117→    try {
   118→      const res = await fetch("/api/findings?limit=15&sort=recent");
   119→      const data = await res.json();
   120→      if (data.success) setFindings(data.findings);
   121→    } catch {
   122→      /* sessiz */
   123→    }
   124→  }, []);
   125→
   126→  const loadComments = useCallback(async () => {
   127→    try {
   128→      const res = await fetch("/api/comments?limit=40");
   129→      if (res.status === 401) return;
   130→      const data = await res.json();
   131→      if (data.success) {
   132→        setComments(data.comments);
   133→        setCommentTotal(data.total ?? data.comments.length);
   134→      }
   135→    } catch {
   136→      /* sessiz */
   137→    }
   138→  }, []);
   139→
   140→  useEffect(() => {
   141→    fetch("/api/admin/session")
   142→      .then((r) => r.json())
   143→      .then((data) => setAuth(data.authenticated ? "admin" : "guest"))
   144→      .catch(() => setAuth("guest"));
   145→  }, []);
   146→
   147→  useEffect(() => {
   148→    if (auth !== "admin") return;
   149→    loadStats();
   150→    loadFindings();
   151→    loadComments();
   152→  }, [auth, loadStats, loadFindings, loadComments]);
   153→
   154→  const login = async (e: React.FormEvent) => {
   155→    e.preventDefault();
   156→    setLoggingIn(true);
   157→    setLoginError(null);
   158→    try {
   159→      const res = await fetch("/api/admin/login", {
   160→        method: "POST",
   161→        headers: { "Content-Type": "application/json" },
   162→        body: JSON.stringify({ password }),
   163→      });
   164→      const data = await res.json();
   165→      if (data.success) {
   166→        setPassword("");
   167→        setAuth("admin");
   168→      } else {
   169→        setLoginError(data.error || "Giriş başarısız.");
   170→      }
   171→    } catch {
   172→      setLoginError("Bağlantı hatası — giriş yapılamadı.");
   173→    } finally {
   174→      setLoggingIn(false);
   175→    }
   176→  };
   177→
   178→  const logout = async () => {
   179→    await fetch("/api/admin/logout", { method: "POST" }).catch(() => {});
   180→    setAuth("guest");
   181→  };
   182→
   183→  // ---- Tarama akışı (yalnızca bu panelde) ----
   184→  const startScan = async () => {
   185→    if (scanning) return;
   186→    setScanning(true);
   187→    setLog([]);
   188→    setSessionAdded(0);
   189→    setDoneLangs(0);
   190→    setCurrentLang(null);
   191→
   192→    try {
   193→      const startRes = await fetch("/api/scan/start", {
   194→        method: "POST",
   195→        headers: { "Content-Type": "application/json" },
   196→        body: JSON.stringify({ scope, focusCategory }),
   197→      });
   198→      const startData = await startRes.json();
   199→      if (!startData.success) throw new Error(startData.error || "Tarama başlatılamadı");
   200→
   201→      const scanId: string = startData.scanId;
   202→      const langs: { code: string; name: string; flag: string }[] = startData.languages;
   203→      setTotalLangs(langs.length);
   204→      let addedTotal = 0;
   205→
   206→      for (let i = 0; i < langs.length; i++) {
   207→        const lang = langs[i];
   208→        setCurrentLang({ flag: lang.flag, name: lang.name });
   209→        const logId = `${lang.code}-${Date.now()}`;
   210→        setLog((prev) => [
   211→          ...prev,
   212→          { id: logId, flag: lang.flag, langName: lang.name, status: "searching" },
   213→        ]);
   214→
   215→        try {
   216→          const res = await fetch("/api/scan/step", {
   217→            method: "POST",
   218→            headers: { "Content-Type": "application/json" },
   219→            body: JSON.stringify({ scanId, langCode: lang.code, focusCategory }),
   220→          });
   221→          const data = await res.json();
   222→          if (data.success) {
   223→            addedTotal += data.added;
   224→            setSessionAdded(addedTotal);
   225→            setLog((prev) =>
   226→              prev.map((e) =>
   227→                e.id === logId
   228→                  ? { ...e, status: "done", searched: data.searched, relevant: data.relevant, added: data.added }
   229→                  : e
   230→              )
   231→            );
   232→          } else {
   233→            setLog((prev) => prev.map((e) => (e.id === logId ? { ...e, status: "error" } : e)));
   234→          }
   235→        } catch {
   236→          setLog((prev) => prev.map((e) => (e.id === logId ? { ...e, status: "error" } : e)));
   237→        }
   238→        setDoneLangs(i + 1);
   239→      }
   240→
   241→      await fetch("/api/scan/complete", {
   242→        method: "POST",
   243→        headers: { "Content-Type": "application/json" },
   244→        body: JSON.stringify({ scanId }),
   245→      });
   246→
   247→      loadFindings();
   248→      loadStats();
   249→      loadComments();
   250→      toast({
   251→        title: "Küresel tarama tamamlandı",
   252→        description: `${langs.length} dil tarandı, ${addedTotal} yeni bulgu Türkçeleştirilerek dergiye eklendi.`,
   253→      });
   254→    } catch {
   255→      toast({
   256→        title: "Tarama başlatılamadı",
   257→        description: "Oturum süresi dolmuş olabilir; tekrar giriş yapmayı deneyin.",
   258→        variant: "destructive",
   259→      });
   260→    } finally {
   261→      setScanning(false);
   262→      setCurrentLang(null);
   263→    }
   264→  };
   265→
   266→  const deleteFinding = async (f: FindingItem) => {
   267→    setDeletingFindingId(f.id);
   268→    try {
   269→      const res = await fetch(`/api/findings/${f.id}`, { method: "DELETE" });
   270→      const data = await res.json();
   271→      if (data.success) {
   272→        setFindings((prev) => prev.filter((x) => x.id !== f.id));
   273→        loadStats();
   274→        toast({ title: "Bulgu silindi", description: f.trTitle.slice(0, 80) });
   275→      } else {
   276→        toast({ title: "Silinemedi", description: data.error, variant: "destructive" });
   277→      }
   278→    } catch {
   279→      toast({ title: "Silinemedi", description: "Bağlantı hatası.", variant: "destructive" });
   280→    } finally {
   281→      setDeletingFindingId(null);
   282→    }
   283→  };
   284→
   285→  const deleteComment = async (c: AdminComment) => {
   286→    setDeletingCommentId(c.id);
   287→    try {
   288→      const res = await fetch(`/api/comments/${c.id}`, { method: "DELETE" });
   289→      const data = await res.json();
   290→      if (data.success) {
   291→        setComments((prev) => prev.filter((x) => x.id !== c.id));
   292→        setCommentTotal((t) => Math.max(0, t - 1));
   293→      } else {
   294→        toast({ title: "Silinemedi", description: data.error, variant: "destructive" });
   295→      }
   296→    } catch {
   297→      toast({ title: "Silinemedi", description: "Bağlantı hatası.", variant: "destructive" });
   298→    } finally {
   299→      setDeletingCommentId(null);
   300→    }
   301→  };
   302→
   303→  // ---- Yeni haber: görsel yükleme + kaydetme ----
   304→  const uploadImage = async (file: File) => {
   305→    setUploading(true);
   306→    try {
   307→      const fd = new FormData();
   308→      fd.append("file", file);
   309→      const res = await fetch("/api/admin/upload", { method: "POST", body: fd });
   310→      const data = await res.json();
   311→      if (data.success && data.url) {
   312→        setNewsImages((prev) =>
   313→          prev.length >= 8
   314→            ? prev
   315→            : [...prev, { url: data.url, alt: "", preview: data.url }]
   316→        );
   317→        toast({ title: "Görsel yüklendi", description: data.url });
   318→      } else {
   319→        toast({ title: "Görsel yüklenemedi", description: data.error, variant: "destructive" });
   320→      }
   321→    } catch {
   322→      toast({ title: "Görsel yüklenemedi", description: "Bağlantı hatası.", variant: "destructive" });
   323→    } finally {
   324→      setUploading(false);
   325→      if (fileInputRef.current) fileInputRef.current.value = "";
   326→    }
   327→  };
   328→
   329→  const onImagePicked = (e: React.ChangeEvent<HTMLInputElement>) => {
   330→    const file = e.target.files?.[0];
   331→    if (file) uploadImage(file);
   332→  };
   333→
   334→  const resetNewsForm = () => {
   335→    setNewsTitle("");
   336→    setNewsBody("");
   337→    setNewsCategory("genel");
   338→    setNewsSource("");
   339→    setNewsImportance("4");
   340→    setNewsNewTech(false);
   341→    setNewsImages([]);
   342→  };
   343→
   344→  const submitNews = async (e: React.FormEvent) => {
   345→    e.preventDefault();
   346→    if (savingNews || uploading) return;
   347→    setSavingNews(true);
   348→    try {
   349→      const res = await fetch("/api/admin/findings", {
   350→        method: "POST",
   351→        headers: { "Content-Type": "application/json" },
   352→        body: JSON.stringify({
   353→          trTitle: newsTitle,
   354→          trSummary: newsBody,
   355→          category: newsCategory,
   356→          sourceName: newsSource,
   357→          importance: Number(newsImportance) || 4,
   358→          isNewTech: newsNewTech,
   359→          images: newsImages.map((im) => ({ url: im.url, alt: im.alt })),
   360→        }),
   361→      });
   362→      const data = await res.json();
   363→      if (data.success) {
   364→        resetNewsForm();
   365→        loadFindings();
   366→        loadStats();
   367→        toast({
   368→          title: "Haber dergiye eklendi",
   369→          description: "Yeni haber, derginin en güncel kaydı olarak okurlara sunuldu.",
   370→        });
   371→      } else {
   372→        toast({ title: "Haber kaydedilemedi", description: data.error, variant: "destructive" });
   373→      }
   374→    } catch {
   375→      toast({ title: "Haber kaydedilemedi", description: "Bağlantı hatası.", variant: "destructive" });
   376→    } finally {
   377→      setSavingNews(false);
   378→    }
   379→  };
   380→
   381→  // ================= GİRİŞ EKRANI =================
   382→  if (auth === "checking") {
   383→    return (
   384→      <div className="flex min-h-screen items-center justify-center bg-slate-950 text-slate-100">
   385→        <Loader2 className="h-6 w-6 animate-spin text-amber-400" />
   386→      </div>
   387→    );
   388→  }
   389→
   390→  if (auth === "guest") {
   391→    return (
   392→      <div className="flex min-h-screen flex-col items-center justify-center gap-6 bg-slate-950 px-4 text-slate-100">
   393→        <div className="flex items-center gap-3">
   394→          <OyiMark className="h-14 w-14" />
   395→          <div>
   396→            <h1 className="mag-serif text-xl font-black tracking-tight">
   397→              OYİ <span className="text-amber-400">GLOBAL</span>
   398→            </h1>
   399→            <p className="text-xs text-slate-400">Yönetim Paneli — Editör Girişi</p>
   400→          </div>
   401→        </div>
   402→
   403→        <form
   404→          onSubmit={login}
   405→          className="w-full max-w-sm space-y-3 rounded-xl border border-slate-800 bg-slate-900/70 p-6 shadow-xl"
   406→        >
   407→          <div className="flex items-center gap-2 text-sm font-semibold text-slate-200">
   408→            <Lock className="h-4 w-4 text-amber-400" />
   409→            Yönetici şifresi
   410→          </div>
   411→          <Input
   412→            type="password"
   413→            value={password}
   414→            onChange={(e) => setPassword(e.target.value)}
   415→            placeholder="Şifre"
   416→            autoFocus
   417→            className="bg-slate-950 border-slate-700"
   418→          />
   419→          {loginError && <p className="text-xs font-semibold text-red-400">{loginError}</p>}
   420→          <Button
   421→            type="submit"
   422→            disabled={loggingIn || password.length === 0}
   423→            className="w-full bg-amber-500 hover:bg-amber-400 text-slate-950 font-bold"
   424→          >
   425→            {loggingIn ? <Loader2 className="h-4 w-4 animate-spin" /> : <ShieldCheck className="h-4 w-4" />}
   426→            Panele Giriş Yap
   427→          </Button>
   428→          <p className="text-[11px] leading-snug text-slate-500">
   429→            Şifre projenin <code className="text-slate-400">.env</code> dosyasındaki{" "}
   430→            <code className="text-amber-400/90">ADMIN_PASSWORD</code> ile belirlenir.
   431→          </p>
   432→        </form>
   433→
   434→        <Link href="/" className="text-xs text-slate-400 underline-offset-2 hover:text-amber-300 hover:underline">
   435→          ← Dergiye geri dön
   436→        </Link>
   437→      </div>
   438→    );
   439→  }
   440→
   441→  // ================= PANEL =================
   442→  const lastScan = stats?.lastScan;
   443→  const lastScanDate = lastScan
   444→    ? new Date(lastScan.startedAt).toLocaleString("tr-TR", {
   445→        day: "numeric",
   446→        month: "short",
   447→        hour: "2-digit",
   448→        minute: "2-digit",
   449→      })
   450→    : null;
   451→
   452→  return (
   453→    <div className="min-h-screen bg-slate-950 text-slate-100">
   454→      {/* Üst bar */}
   455→      <header className="sticky top-0 z-30 border-b border-slate-800/80 bg-slate-950/90 backdrop-blur-md">
   456→        <div className="mx-auto flex h-16 max-w-6xl items-center justify-between gap-3 px-4 sm:px-6">
   457→          <div className="flex items-center gap-3 min-w-0">
   458→            <OyiMark className="h-10 w-10 shrink-0" />
   459→            <div className="min-w-0">
   460→              <h1 className="mag-serif font-black text-base leading-tight tracking-tight text-slate-50">
   461→                OYİ <span className="text-amber-400">GLOBAL</span>
   462→                <span className="ml-2 rounded-sm border border-emerald-500/40 bg-emerald-500/10 px-1.5 py-0.5 align-middle text-[9px] font-bold uppercase tracking-[0.18em] text-emerald-300">
   463→                  Yönetim
   464→                </span>
   465→              </h1>
   466→              <p className="text-[11px] text-slate-400 truncate">
   467→                Dergi taraması, içerik ve yorum yönetimi
   468→              </p>
   469→            </div>
   470→          </div>
   471→          <div className="flex items-center gap-2">
   472→            <Link
   473→              href="/"
   474→              className="inline-flex h-9 items-center gap-1.5 rounded-md border border-slate-700 bg-slate-900 px-3 text-sm text-slate-300 transition-colors hover:border-amber-500/50 hover:text-amber-300"
   475→            >
   476→              <ExternalLink className="h-3.5 w-3.5" />
   477→              <span className="hidden sm:inline">Dergiye Git</span>
   478→            </Link>
   479→            <Button
   480→              variant="outline"
   481→              onClick={logout}
   482→              className="h-9 border-slate-700 bg-slate-900 px-3 text-slate-300 hover:bg-slate-800 hover:text-red-300"
   483→            >
   484→              <LogOut className="h-3.5 w-3.5" />
   485→              <span className="hidden sm:inline">Çıkış</span>
   486→            </Button>
   487→          </div>
   488→        </div>
   489→      </header>
   490→
   491→      <main className="mx-auto w-full max-w-6xl space-y-6 px-4 py-6 sm:px-6">
   492→        {/* İstatistikler */}
   493→        <section className="grid grid-cols-2 lg:grid-cols-5 gap-3">
   494→          {[
   495→            { icon: Database, label: "Toplam Bulgu", value: stats?.total ?? 0, color: "text-amber-400" },
   496→            { icon: Sparkles, label: "Yeni Teknoloji", value: stats?.newTech ?? 0, color: "text-emerald-400" },
   497→            { icon: Languages, label: "Kaynak Dil", value: stats?.languages.length ?? 0, color: "text-purple-400" },
   498→            { icon: MessageSquare, label: "Okur Yorumu", value: commentTotal, color: "text-sky-400" },
   499→            { icon: ShieldCheck, label: "Son Güncelleme", value: lastScanDate ?? "—", color: "text-slate-200", small: true },
   500→          ].map((s, i) => (
   501→            <Card key={i} className="border-slate-800 bg-slate-900/70">
   502→              <CardContent className="p-4 flex items-center gap-3">
   503→                <div className="h-9 w-9 rounded-lg bg-slate-800/80 border border-slate-700 flex items-center justify-center shrink-0">
   504→                  <s.icon className={`h-4 w-4 ${s.color}`} />
   505→                </div>
   506→                <div className="min-w-0">
   507→                  <p className={`font-bold leading-none ${s.small ? "text-[13px] truncate" : "text-xl"}`}>
   508→                    {s.value}
   509→                  </p>
   510→                  <p className="text-[11px] text-slate-400 mt-1">{s.label}</p>
   511→                </div>
   512→              </CardContent>
   513→            </Card>
   514→          ))}
   515→        </section>
   516→
   517→        {/* Tarama paneli */}
   518→        <section className="rounded-2xl border border-slate-800 bg-gradient-to-b from-slate-900 to-slate-950 p-5 sm:p-6 space-y-4">
   519→          <div className="flex flex-wrap items-center justify-between gap-2">
   520→            <h2 className="flex items-center gap-2 font-semibold text-slate-100">
   521→              <Radar className="h-4 w-4 text-amber-400" />
   522→              Küresel Tarama — Yalnızca Editör
   523→            </h2>
   524→            <Badge className="bg-emerald-500/15 text-emerald-300 border-emerald-500/30 gap-1 hover:bg-emerald-500/15">
   525→              <ShieldCheck className="h-3 w-3" /> Admin yetkisi aktif
